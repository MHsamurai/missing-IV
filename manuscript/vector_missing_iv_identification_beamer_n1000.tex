\section{Motivation}
\begin{frame}{Missing-data mechanisms and standard approaches}
Let $D$ be the response pattern; $Y=(Y_{\mathrm{obs}},Y_{\mathrm{mis}})$.
\smallskip
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{@{}p{.19\linewidth}p{.40\linewidth}p{.36\linewidth}@{}}
\toprule
Mechanism & Response probability & Common approaches\\
\midrule
MCAR & $\Prb(D\mid Y)=\Prb(D)$ & Complete-case analysis\\[.5em]
MAR & $\Prb(D\mid Y)=\Prb(D\mid Y_{\mathrm{obs}})$ & Multiple imputation; IPW\\[.5em]
MNAR & May depend on $Y_{\mathrm{mis}}$ even given $Y_{\mathrm{obs}}$ & Selection / pattern-mixture models; \redkey{Missing IV}\\
\bottomrule
\end{tabular}
\medskip
MCAR: missing completely at random; MAR: missing at random; MNAR: missing not at random.\vspace{-1pt}
\begin{takeaway}
Under MNAR, respondent-only analysis can induce selection bias. A \redkey{Missing IV / shadow variable} can identify the full-data law under suitable assumptions.
\end{takeaway}
{\slideannotation Little and Rubin (2002); Heckman (1979, \emph{Econometrica}).}
\end{frame}

\begin{frame}{Latent measurement model}
\begin{columns}[c,onlytextwidth]
\begin{column}{.48\textwidth}
\centering
\input{manuscript/figures/readable/latent_measurement_model.tex}
\end{column}
\begin{column}{.49\textwidth}
In structural equation modeling (SEM) and item response theory (IRT), \term{latent} $F$ underlies measurements $Y=(Y_1,\ldots,Y_m)$.
\begin{itemize}
  \item $Q_F(\cdot\mid W,X)$ is the latent distribution.
  \item $Q_j(\cdot\mid F,X)$ is the item measurement kernel.
  \item Items are conditionally independent given $(F,X)$.
\end{itemize}
\end{column}
\end{columns}
\vspace{.2em}
\[
\Prb(Y=y\mid W=w,X=x)
=\int\prod_{j=1}^m Q_j(y_j\mid f,x)\,dQ_F(f\mid w,x).
\]
{\footnotesize
Allman et al. (2009, \emph{AoS}) \textbf{identify finite class proportions and measurement kernels} up to labels under three-view rank conditions.\par
}
\end{frame}

\begin{frame}{Lemma 2.1}
\small
Let $U=(X,W)$, $R=\prod_{j=1}^mD_j$, and $P_1=\Prb(Y,U\mid R=1)$.
$U,Z,R$ are always observed; the entire vector $Y$ is observed when $R=1$.
\begin{theorembox}{Global complete-case recovery}
Assume $R\indep Z\mid(Y,U)$, $\pi(y,u)=\Prb(R=1\mid y,u)>0$, and $\pi^{-1}\in L^2(P_1)$.
For every $h\in L^2(P_1)$, assume
\[
\E\{h(Y,U)\mid R=1,Z,U\}=0\ \Longrightarrow\ h=0\quad P_1\text{-a.s.}
\]
Then $\pi$ uniquely solves, among positive $q$ with $q^{-1}\in L^2(P_1)$,
\[
\E\{R/q(Y,U)\mid Z,U\}=1,
\]
For every integrable $h$, $\E h(Y,Z,U)=\E\{Rh(Y,Z,U)/\pi(Y,U)\}$ identifies the full law.
\end{theorembox}
\bluekey{A direct vector extension, but it requires global complete-case positivity.}
\par{\footnotesize d'Haultfoeuille (2010, \emph{JoE}); Miao et al. (2019, arXiv).}
\end{frame}

\begin{frame}{Assumption 5}
\begin{block}{Supported block}
A subset $S\subseteq\{1,\ldots,m\}$ is \term{supported} when all measurements in $Y_S$ are jointly observed with positive probability on the target support. A global complete case is not required.
\end{block}
\begin{columns}[c,onlytextwidth]
\begin{column}{.53\textwidth}
For a block $S$, define
\[
R_S=\prod_{j\in S}D_j,\quad Y_S=(Y_j:j\in S).
\]
$U_S=(X,W)$. The support condition is
\[
\pi_S(y,u):=\Prb(R_S=1\mid y,u)>0.
\]
$U_S,Z_S,R_S$ are always observed. Use $Y_S$ as a complete block only when $R_S=1$.
\end{column}
\begin{column}{.44\textwidth}
\centering
\input{manuscript/figures/readable/supported_block_set_diagram.tex}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Stage 1 (1): Recover the block law}
\bluekey{Recover the population law, including people with missing items.}\par
\smallskip
Let $Y_S$ be the items in block $S$, $R_S=1$ indicate that all are observed,
and $U_S=(X,W)$ be observed covariates.
\begin{block}{\textcolor{Red}{New!} Proposition 1: Key result}
With Missing IV $Z_S$, positivity and complete-case completeness,
\[
\E\left\{\frac{R_S}{\pi_S(Y_S,U_S)}\middle|Z_S,U_S\right\}=1,
\qquad \pi_S(y,u)=\Prb(R_S=1\mid Y_S=y,U_S=u).
\]
The moment identifies $\pi_S$ in the stated bridge class, hence the
block law $\Prb(Y_S,Z_S,U_S)$ is identified.
\end{block}
\begin{itemize}
\item Weight observed blocks by $1/\pi_S$ to correct MNAR selection.
\item \redkey{No parametric selection model}; exclusion and completeness are required.
\end{itemize}
{\slideannotation Assumptions 1, 5--7 and $\pi_S^{-1}\in L^2\{P(Y_S,U_S\mid R_S=1)\}$; details in Appendix.}
\end{frame}

\begin{frame}{Stage 1 (2): Use overlapping pairs}
\bluekey{We need the required pairs to be observable, not all items at once.}
\begin{block}{Supported blocks}
Each required block has positive observation probability on its target support.
Use anchor pair $\{a,b\}$ and extension pairs $\{j,a\}$:
\[
\underbrace{(Y_a,Y_b,W)}_{\text{anchor law}}
\qquad\longleftrightarrow\qquad
\underbrace{(Y_j,Y_a,W)}_{\text{extension law for each }j}.
\]
\end{block}
\begin{itemize}
\item Apply Stage 1 separately to each pair, conditional on $X$.
\item The shared item $Y_a$ connects the recovered pair laws.
\item $Z_S$ corrects missingness; $W$ supplies variation for latent decomposition.
\end{itemize}
\medskip
\begin{takeaway}
\bluekey{Output: overlapping pair laws.}\quad
\redkey{Global complete cases are not required.}
\end{takeaway}
\end{frame}

\begin{frame}{Stage 2 (1): Identify the latent structure}
\bluekey{Decompose one anchor tensor, then recover every extension item.}\par
\smallskip
Let $F\in\{1,\ldots,r\}$ denote the latent class,
$M_j(y,f;x)=\Prb(Y_j=y\mid F=f,X=x)$, and
$\lambda_f(w,x)=\Prb(F=f\mid W=w,X=x)$.
\begin{block}{\textcolor{Red}{New!} Theorem 1: Key result}
The recovered law of $(Y_a,Y_b,W)\mid X$ identifies the anchor factors.
Overlapping extension laws then identify
\[
\underbrace{Q_F=\{\lambda_f(w,x)\}}_{\text{latent-class distribution}}
\qquad\text{and}\qquad
\underbrace{Q_Y=\{M_j(y,f;x)\}_j}_{\text{all measurement kernels}}.
\]
\end{block}
\begin{itemize}
\item Allman et al. (2009, \emph{AoS}): three views, with $W$ as the third mode.
\item Known finite $r$, finite supports, local independence and measurement exclusion.
\item Three-view ranks satisfy $k_a+k_b+k_G\ge 2r+2$ (Assumption 3.1).
\item Anchor ordering fixes \redkey{common labels}; otherwise identification is up to permutation.
\end{itemize}
\end{frame}

\begin{frame}{Stage 2 (2): Connect identification to estimation}
\begin{block}{Recover the joint outcome law}
For any item set $A$, including items never jointly observed,
\[
\Prb(Y_A=y_A\mid W=w,X=x)
=\sum_{f=1}^r\lambda_f(w,x)\prod_{j\in A}M_j(y_j,f;x).
\]
\end{block}
\begin{itemize}
\item \textbf{Stage 1:} estimate each pair's inverse observation probability.
\item \textbf{Stage 2:} fit a \term{bridge-weighted composite criterion} to the observed pairs.
\end{itemize}
\medskip
\begin{takeaway}
\bluekey{New: connect MNAR block-law recovery and latent decomposition in one identification theorem.}
\end{takeaway}
\medskip
\textbf{Next:} finite-dimensional inference under Appendix regularity conditions;
account for bridge estimation.
\end{frame}

\begin{frame}{\textcolor{Red}{New!} Theorem 5.1}
\begin{theorembox}{Consistency and asymptotic normality}
Under Assumption 5.1 and Proposition 4.1,
\[
\widehat\theta\to_p\theta_0,\qquad
\sqrt n(\widehat\theta-\theta_0)\rightsquigarrow N(0,A^{-1}BA^{-\top}).
\]
Let $\psi$ denote the composite score. At $(\theta_0,\eta_0)$,
\[
A=\E(\partial_\theta\psi),\qquad C=\E(\partial_\eta\psi),\qquad
B=\operatorname{Var}\{\psi+C\varphi_\eta\}.
\]
\end{theorembox}
\begin{itemize}
  \item The sandwich variance includes \textbf{first-stage bridge estimation}.
  \item Individual-level bootstrap repeats both stages and label alignment.
  \item A growing sieve needs additional rate and inverse-problem conditions.
\end{itemize}
\end{frame}

\begin{frame}{Boundary estimates limit Wald inference}
\begin{itemize}
\setlength{\itemsep}{.65em}
\item Candidate: MNAR boundary $19/100$, valid CI $81/100$;
MCAR/MAR boundary $6/100$ each.
\item Restricted selection: MNAR boundary $100/100$; no Wald SE or coverage.
\item Boundary means a fitted logit has absolute value at least $9.9$
under bounds $[-10,10]$.
\item CI screening also checks convergence, gradient, and positive-definite Hessian.
SE uses sandwich + delta, including empirical $P(V)$.
\item \term{Valid-only coverage} and \term{covered-and-available rate}
have different denominators. Neither repairs missing intervals.
\end{itemize}
\medskip
\bluekey{Population identification is not a finite-sample CI guarantee.}
\par\smallskip
Weak rank, boundary behavior, global optimization, and formal candidate
asymptotics remain separate issues. 100 runs also have Monte Carlo error.
\end{frame}

\begin{frame}{Conclusion}
\begin{enumerate}
\setlength{\itemsep}{.85em}
\item \bluekey{Missingness can be part of the observed category tensor.}
Under explicit restrictions, three-view uniqueness plus normalization
identifies the response cells, measurement kernels, and latent distribution.
\item \bluekey{The new MNAR experiment shows bias correction at $n=1{,}000$.}
Saturated selection is equivalent to correctly specified saturated logistic;
it is not superior to a distinct correct parametric method.
\item \bluekey{Inference and generalization still require work.}
Boundary estimates, unknown classes, direct response effects of $V/F$,
dependent responses, and absent complete cases are not resolved here.
\end{enumerate}
\medskip
Novelty relative to latent-class selection models and a candidate-specific
asymptotic theorem are not established by this experiment.
\end{frame}

\begin{frame}{Appendix: proof sketch of three-view uniqueness}
Given $F=f,Y_1=a$, joint local independence and response exclusions imply
$T_2\indep T_3\indep V$ and
\[
K_{jf}^{(a)}(y)=\pi_j(a,y)M_j(y,f),\quad
K_{jf}^{(a)}(\mathrm{mis})=1-\sum_{y=0}^1\pi_j(a,y)M_j(y,f).
\]
\begin{itemize}
\setlength{\itemsep}{.5em}
\item Positive $\pi_j$ and class contrast make the two $K_j$ columns distinct.
\item Nonconstant $\lambda(v)$ makes the two probability columns of $G$ distinct.
\item Distinct normalized two-column probability matrices have rank 2.
Thus column Kruskal ranks sum to $6=2r+2$.
\item Allman et al.'s uniqueness result identifies factors up to scaling/labels;
probability normalization removes scaling.
\end{itemize}
\bluekey{$G=P(V\mid F)$ is a Bayes representation, not a reversed generating DAG.}
\par\smallskip
{\slideannotation Allman, Matias and Rhodes (2009), \emph{Annals of Statistics} 37, 3099--3132, Section 4, Theorem 1 and Corollary 2. doi:10.1214/09-AOS689.}
\end{frame}

\begin{frame}{Appendix: separate response, align labels, recover the law}
For each $j=2,3$ and $a$, solve for $r_j(a,y)=1/\pi_j(a,y)$:
\[
\begin{pmatrix}
K_{j1}^{(a)}(0)&K_{j1}^{(a)}(1)\\
K_{j2}^{(a)}(0)&K_{j2}^{(a)}(1)
\end{pmatrix}
\begin{pmatrix}r_j(a,0)\\r_j(a,1)\end{pmatrix}
=\begin{pmatrix}1\\1\end{pmatrix}.
\]
Its determinant is
$\pi_j(a,0)\pi_j(a,1)\{M_j(1,2)-M_j(1,1)\}\ne0$.
Recover $M_j(y,f)=K_{jf}^{(a)}(y)r_j(a,y)$; align strata by the $M_2$ order.
\[
p_f=\sum_aP(Y_1=a)\omega_f(a),\qquad
M_1(a,f)=\frac{P(Y_1=a)\omega_f(a)}{p_f},
\]
\[
P(F=f\mid V=v)=\frac{p_fG_f(v)}{P(v)},\qquad
P(Y=y)=\sum_f p_f\prod_{j=1}^3M_j(y_j,f).
\]
\bluekey{Population inversion is unique; unconstrained sample inversion need not be valid.}
\end{frame}
