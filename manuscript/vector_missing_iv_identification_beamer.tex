\begin{frame}{Missing-data mechanisms and standard approaches}
\fontsize{9pt}{10.5pt}\selectfont
\begin{block}{Definition}
Let $D$ denote the response pattern and write $Y=(Y_{\mathrm{obs}},Y_{\mathrm{mis}})$. Little and Rubin (2002) distinguish missing completely at random (MCAR), missing at random (MAR), and missing not at random (MNAR). An instrumental variable for missingness is termed a \textcolor{Red}{\bfseries Missing IV or shadow variable}.
\end{block}
\vspace{.1em}
\setlength{\tabcolsep}{3pt}
\renewcommand{\arraystretch}{1.0}
\begin{tabular}{>{\raggedright\arraybackslash}p{.13\linewidth}
                >{\raggedright\arraybackslash}p{.45\linewidth}
                >{\raggedright\arraybackslash}p{.32\linewidth}}
\toprule
Mechanism & Assumption & Common approaches \\
\midrule
MCAR & Response does not depend on observed or missing data. $\Prb(D\mid Y)=\Prb(D)$ & Complete-case analysis, mean imputation \\
\addlinespace[.15em]
MAR & Response depends only on observed data. $\Prb(D\mid Y)=\Prb(D\mid Y_{\mathrm{obs}})$ & Multiple imputation, inverse probability weighting \\
\addlinespace[.15em]
MNAR & Response also depends on missing data. $\Prb(D\mid Y)=\Prb(D\mid Y_{\mathrm{obs}},Y_{\mathrm{mis}})$ & Selection models, pattern-mixture models, \textcolor{Red}{\bfseries Missing IV} \\
\bottomrule
\end{tabular}
\vspace{.15em}
\begin{takeaway}
Under MNAR, respondent-only analysis can induce sample-selection bias (Heckman, 1979). Missing IVs recover full-data laws before latent decomposition.
\end{takeaway}
\end{frame}

\begin{frame}{Usual IV and Missing IV}
Let $R$ indicate that $Y$ is observed and let $Z$ be an always-observed auxiliary variable.

\begin{block}{Missing IV}
\centering
\begin{tikzpicture}[>=Latex,node distance=1.05cm,
  var/.style={circle,draw=Navy,fill=white,minimum size=.78cm,thick},

  \node[text=Red,font=\bfseries] at ($(z)!.5!(r)+(0,-.72)$) {$Z\nrightarrow R$};

\par\smallskip
$Z$ predicts $Y$ and satisfies $R\indep Z\mid Y$.
\end{block}
\end{column}
\end{columns}
\vspace{.15em}
For a Missing IV, positivity gives
\[
\E\left\{\frac{R}{\pi(Y)}\middle|Z\right\}=1,
\qquad \pi(y)=\Prb(R=1\mid Y=y),
\]
and completeness makes $\pi$ unique (d'Haultfoeuille, 2010, \emph{JoE}).
\end{frame}

\begin{frame}{Latent measurement model}
\begin{columns}[c,onlytextwidth]
\begin{column}{.48\textwidth}
\centering
\resizebox{.98\linewidth}{!}{\input{notes/figures/latent_measurement_model.tex}}
\end{column}
\begin{column}{.49\textwidth}
\small
This structure underlies structural equation modeling (SEM) and item response theory (IRT).
\normalsize
Let $F$ denote an \term{unobserved latent variable} and let $Y=(Y_1,\ldots,Y_m)$ denote its potential \term{observed measurements}.
\begin{itemize}
  \item $Q_F(\cdot\mid W,X)$ is the latent distribution.
  \item $Q_j(\cdot\mid F,X)$ is the measurement kernel for item $j$.
  \item $Y_1,\ldots,Y_m$ are conditionally independent given $(F,X)$.
\end{itemize}
\end{column}
\end{columns}
\vspace{.2em}
\[
\Prb(Y=y\mid W=w,X=x)
=\int\prod_{j=1}^m Q_j(y_j\mid f,x)\,dQ_F(f\mid w,x).
\]
\begin{takeaway}
With complete data, three sufficiently informative views identify finite class proportions and measurement kernels up to labels (Allman et al., 2009).
\end{takeaway}
\end{frame}

\begin{frame}{A latent shifter is not a Missing IV}
Suppose $D\indep W\mid(Y,F,X)$. After integrating out $F$,
\[
\Prb(D=d\mid Y=y,W=w,X=x)
=\int \Prb(D=d\mid y,f,x)\,d\Prb(f\mid y,w,x).
\]
If $W$ shifts $F$, the mixing distribution generally depends on $w$. Hence
\[
D\indep W\mid(Y,X)\quad\text{does }\textcolor{Red}{\mathbf{not}}\text{ follow.}
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{$Z_S$: Missing IV}
Predicts $Y_S$ and satisfies an exclusion for $R_S$ conditional on $(Y_S,U_S)$.
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\begin{block}{$W$: latent shifter}
Moves the distribution of $F$ and supplies third-mode variation for decomposition.
\end{block}
\end{column}
\end{columns}
\begin{takeaway}
Recovering a block law does not itself separate $Q_F$ from $Q_Y$; $W$ supplies a decomposition mode, not a Missing IV.
\end{takeaway}
\end{frame}

\begin{frame}{Assumption 1}
\begin{block}{Observation process}
Let $Y=(Y_1,\ldots,Y_m)$ denote the measurement vector and let
$D=(D_1,\ldots,D_m)\in\{0,1\}^m$ denote its response indicators.
\begin{itemize}
  \item $Y_j$ is observed if and only if $D_j=1$.
  \item $X,W,Z,D$ are always observed; the latent variable $F$ is not.
  \item $Z$ collects the block-specific Missing IVs $Z_S$.
\end{itemize}
The \term{observed data for one individual} are
\[
O=(X,W,Z,D,Y_D),\qquad Y_D=\{Y_j:D_j=1\}.
\]
\end{block}
\bluekey{$O$ records the response pattern and only the measurements actually observed.}
\end{frame}

\begin{frame}{Assumptions 2 and 3}
\begin{block}{Assumption 2: Latent-distribution shift}
Let $X$ denote ordinary covariates and $W$ the \term{latent shifter}.
The conditional latent distribution, with $Z$ marginalized, is
\[
F\mid(W,X)\sim Q_F(\cdot\mid W,X).
\]
\end{block}
\begin{block}{Assumption 3: Measurement process}
Let $Q_j$ denote the \term{measurement kernel} of item $j$ and
$\theta_j$ its parameter. For $j=1,\ldots,m$,
\[
Y_j\mid(F,X)\sim Q_j(\cdot\mid F,X;\theta_j).
\]
\end{block}
\bluekey{$Q_F$ describes who is in each latent class; $Q_j$ describes what each class measures.}
\end{frame}

\begin{frame}{Assumption 4}
\begin{block}{Local independence and measurement exclusion}
With $Z$ marginalized, the \textbf{joint} measurement law satisfies
\[
P(Y\mid F,W,X)=\prod_{j=1}^m Q_j(Y_j\mid F,X).
\]
\begin{itemize}
  \item All items are mutually conditionally independent given $(F,X)$.
  \item $W$ shifts $Q_F$ but does not directly enter the measurement kernels.
  \item This is a restriction on the full item vector, not merely pairwise independence.
\end{itemize}
\end{block}
\begin{takeaway}
We do not assume $Y\indep Z\mid(F,W,X)$. Missing-IV exclusion is a separate restriction on the response indicator (Assumption 6).
\end{takeaway}
\end{frame}

\begin{frame}{A simple multivariate extension}
Let $D=(D_1,\ldots,D_m)'$, $Y=(Y_1,\ldots,Y_m)'$, and $Z$ denote the response indicators, multivariate outcomes, and missing IVs. Define the \term{global complete-case indicator}
\[
R=\mathbf 1\{D=\mathbf 1\}=\prod_{j=1}^m D_j.
\]
If
\[
R\indep Z\mid Y,\qquad \Prb(R=1\mid Y)>0,
\]
and $Y$ is complete with respect to $Z$, then
\[
\E\left\{\frac{R}{\pi(Y)}\middle|Z\right\}=1
\]
uniquely identifies $\pi(y)=\Prb(R=1\mid Y=y)$ and hence the full-data law of $Y$.
\begin{takeaway}
This direct extension is valid, but it relies on observing the entire vector $Y$ in a global complete case.
\end{takeaway}
\end{frame}

\begin{frame}{Assumption 5}
\begin{block}{Supported block}
A subset $S\subseteq\{1,\ldots,m\}$ is \term{supported} when all measurements in $Y_S$ are jointly observed with positive probability on the target support. A global complete case is not required.
\end{block}
\begin{columns}[c,onlytextwidth]
\begin{column}{.53\textwidth}
For a block $S$, define
\[
R_S=\prod_{j\in S}D_j,\quad
Y_S=(Y_j:j\in S),\quad U_S=(X,W).
\]
The support condition is
\[
\pi_S(y,u):=\Prb(R_S=1\mid Y_S=y,U_S=u)>0.
\]
$U_S,Z_S,R_S$ are always observed. The entire $Y_S$ is available when $R_S=1$; a partially observed block is not used as a complete block.
\end{column}

\end{columns}
\end{frame}

\begin{frame}{Assumptions 6 and 7}
\begin{block}{Assumption 6: Blockwise covariate-adjusted Missing IV}
For every supported block $S$,
\[
R_S\indep Z_S\mid(Y_S,U_S).
\]
\end{block}
\begin{block}{Assumption 7: Complete-case completeness}
For $h\in L^2\{\Prb(Y_S,U_S\mid R_S=1)\}$,
\[
\E\{h(Y_S,U_S)\mid R_S=1,Z_S,U_S\}=0
\Longrightarrow h(Y_S,U_S)=0.
\]
The conclusion holds almost surely on $\{R_S=1\}$.
\end{block}
Positivity transfers null sets between the complete-case and full laws; completeness makes the observed conditional operator injective.
\end{frame}

\begin{frame}{\textcolor{Red}{New!} Proposition 1}
\small
\begin{theorembox}{Supported-block identification}
Under blockwise exclusion, positivity, and complete-case completeness, assume
$\pi_S^{-1}\in L^2\{\Prb(Y_S,U_S\mid R_S=1)\}$. Then $\pi_S$ is the unique positive solution with
$q^{-1}\in L^2\{\Prb(Y_S,U_S\mid R_S=1)\}$ of
\[
\E\left\{\frac{R_S}{q(Y_S,U_S)}\middle|Z_S,U_S\right\}=1.
\]
Consequently, for every integrable $h$,
\[
\E h(Y_S,Z_S,U_S)
=\E\left\{\frac{R_Sh(Y_S,Z_S,U_S)}{\pi_S(Y_S,U_S)}\right\},
\]
and $\Prb(Y_S,Z_S,U_S)$ is identified.
\end{theorembox}
\begin{takeaway}
The proposition proves uniqueness within the stated bridge class; the true propensity supplies existence.
\end{takeaway}
\end{frame}

\begin{frame}{Assumption 3.1: Support and model}
\begin{block}{Covariate-assisted anchor-pair structure (1/2)}
Let $F\in\{1,\ldots,r\}$ with known $r\ge2$, and let the item spaces
$\mathcal Y_j$ and the support $\mathcal W_x$ of $W\mid X=x$ be finite.
For every $x$ in a common set $\mathcal X_0$ with $P_X(\mathcal X_0)=1$:
\begin{itemize}
  \item Assumptions 2--4 hold; $p_f(x)>0$ for every class, and $P(W=w\mid X=x)>0$ for every $w\in\mathcal W_x$.
  \item The anchor pair $\{a,b\}$ and every extension pair $\{j,a\}$, $j\notin\{a,b\}$, satisfy Assumptions 1 and 5--7 for every $w$.
  \item Each required pair satisfies the bridge-class condition of Proposition 1:
  \[
  \pi_S^{-1}\in L^2\{P(Y_S,U_S\mid R_S=1)\}.
  \]
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Assumption 3.1: Rank and labels}
\bluekey{The same three-view Kruskal condition as Allman et al. (2009, Theorem 1).}
\begin{block}{Covariate-assisted anchor-pair structure (2/2)}
Let $k_a,k_b,k_G$ be the column Kruskal ranks of $M_a,M_b,G$.
For every $x\in\mathcal X_0$,
\[
\textcolor{Red}{\bm{k_a+k_b+k_G\ge 2r+2}}.
\]
For a known anchor score $s_a$, define
\[
\mu_{af}(x)=\sum_y s_a(y)M_a(y,f;x).
\]
The same strict ordering $\mu_{a1}(x)<\cdots<\mu_{ar}(x)$ holds throughout $\mathcal X_0$.
\end{block}
\bluekey{The rank condition identifies the factors; anchor ordering fixes a common class label.}
\end{frame}

\begin{frame}{Lemma 3.1}
\begin{theorembox}{Khatri--Rao rank}
Let $A$ and $B$ have $r$ nonzero columns, with column Kruskal ranks $k_A$ and $k_B$. Then
\[
k_{B\odot A}\ge \min\{k_A+k_B-1,r\}.
\]
In particular,
\[
\redkey{$k_A+k_B\ge r+1$}
\quad\Longrightarrow\quad
\bluekey{$B\odot A$ has full column rank.}
\]
\end{theorembox}
\begin{takeaway}
This lemma converts the anchor-tensor rank condition into the linear-inversion rank needed for every extension item.
\end{takeaway}
\end{frame}

\begin{frame}{\textcolor{Red}{New!} Theorem 1}
\begin{theorembox}{Anchor-pair identification with $W$ as the third mode}
Under the stage-1 assumptions and the anchor-pair conditions, the observed-data law identifies, under the fixed anchor ordering,
\[
p(x),\quad G(x),\quad
Q_F(x)=\{\lambda_f(w,x)\}_{f,w},\quad
Q_Y(x)=\{M_j(y,f;x)\}_{j,y,f}.
\]
Moreover, for any set of items $A$,
\[
\Prb(Y_A=y_A\mid W=w,X=x)
=\sum_{f=1}^r\lambda_f(w,x)\prod_{j\in A}M_j(y_j,f;x)
\]
is identified even if all items in $A$ are never jointly observed.
\end{theorembox}
\vspace{.2em}
\bluekey{In short: recover the pair laws, decompose one anchor tensor, and obtain the latent-class distribution and every measurement kernel.}
\end{frame}

\begin{frame}{Bridge-weighted composite estimation}
\small
Stage 1 follows d'Haultfoeuille (2010, Section 3): solve the conditional moment equation blockwise and recover each pair law by IPW. Stage 2 follows the plug-in order of Zhao and Shao (2015, Sections 2.2--2.3), while fitting one latent mixture to all overlapping pair laws.
\begin{columns}[T,onlytextwidth]
\begin{column}{.57\textwidth}
Let $\mathcal S_*=\{\{a,b\}\}\cup\{\{j,a\}:j\notin\{a,b\}\}$ and
\[
p_\theta(y_S,w\mid x)=\sum_{f=1}^r p_f(x)G(w,f;x)
\prod_{j\in S}M_j(y_j,f;x).
\]
The proposed estimator is
\[
\widehat\theta=\arg\max_{\theta\in\Theta}
\frac1n\sum_{i=1}^n\sum_{S\in\mathcal S_*}c_S
\frac{R_{iS}\log p_\theta(Y_{iS},W_i\mid X_i)}
{\widehat\pi_S(Y_{iS},U_i)}.
\]
\end{column}
\begin{column}{.40\textwidth}
\begin{itemize}
  \item \term{Composite M-estimator}, not a full likelihood.
  \item The theorem covers a finite-dimensional bridge.
  \item A growing sieve requires additional inverse-problem theory.
  \item $c_S>0$ balances contributions across blocks.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Conditions for estimation}
\begin{block}{Proposition 4.1: A unique population maximizer}
\begin{itemize}
  \item Assumptions 1--7 and 3.1 hold, including anchor ordering.
  \item $\theta_0\in\Theta$ correctly generates the latent measurement model.
  \item For each required pair, $p_\theta(y_S,w\mid x)>0$ on a common support.
  \item The population criterion uses the true bridge and fixed weights $c_S>0$.
\end{itemize}
Then the bridge-weighted population criterion has the unique maximizer $\theta_0$.
\end{block}
\bluekey{Bridge weighting recovers full pair expectations; zero KL loss fixes every pair law, and Theorem 1 fixes the latent parameters.}
\end{frame}

\begin{frame}{Assumption 5.1: Regularity}
Let $O_1,\ldots,O_n$ be i.i.d. and write $\beta=(\theta,\eta)$,
where $\eta=(\eta_S:S\in\mathcal S_*)$ contains the bridge parameters.
Use free coordinates after removing simplex constraints.
\begin{block}{Regularity and separation (1/3)}
\begin{itemize}
  \item $\theta,\eta$ are finite-dimensional and correctly specified.
  \item Their parameter spaces are compact; the true parameters are interior points.
  \item The first-stage population moment has the unique zero $\eta_0$.
  \item The criterion and first-stage moments are twice continuously differentiable near the truth, with square-integrable envelopes.
  \item The bridge Jacobian, composite-score Jacobian, and corresponding Godambe matrix are nonsingular.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Assumption 5.1: Uniform convergence}
\begin{block}{Regularity and separation (2/3)}
Let $\widehat Q_n(\theta,\widehat\eta)$ denote the plug-in criterion and
$Q_0(\theta)$ its population version with the true bridge.
\[
\sup_{\theta\in\Theta}
|\widehat Q_n(\theta,\widehat\eta)-Q_0(\theta)|\xrightarrow{p}0.
\]
The population maximum is separated: for every $\varepsilon>0$ with a nonempty constraint set,
\[
\sup_{\theta\in\Theta:\|\theta-\theta_0\|\ge\varepsilon}Q_0(\theta)
<Q_0(\theta_0).
\]
\end{block}
\bluekey{These global conditions justify consistency; local smoothness alone does not.}
\end{frame}

\begin{frame}{Assumption 5.1: First stage and separation}
\begin{block}{Regularity and separation (3/3)}
The first-stage estimator is regular and asymptotically linear:
\[
\sqrt n(\widehat\eta-\eta_0)
=n^{-1/2}\sum_{i=1}^n\varphi_\eta(O_i)+o_p(1),\quad
\E\varphi_\eta=0,\quad \E\|\varphi_\eta\|^2<\infty.
\]
The following are uniformly bounded away from zero:
\begin{itemize}
  \item class probabilities and supported-pair propensities;
  \item the smallest nonzero singular value of the anchor tensor's $W$-mode unfolding (rows: $W$; columns: $(Y_a,Y_b)$);
  \item the minors supporting the Kruskal-rank conditions;
  \item adjacent gaps in the anchor ordering.
\end{itemize}
\end{block}
\bluekey{Identification alone does not imply regular estimation or stable inference.}
\end{frame}

\begin{frame}{Theorem 5.1}
\begin{theorembox}{Consistency and asymptotic normality}
Under Assumption 5.1 and Proposition 4.1,
\[
\widehat\theta\to_p\theta_0,\qquad
\sqrt n(\widehat\theta-\theta_0)\rightsquigarrow N(0,A^{-1}BA^{-\top}).
\]
Let $\psi$ denote the composite score. At $(\theta_0,\eta_0)$,
\[
A=\E(\partial_\theta\psi),\qquad C=\E(\partial_\eta\psi),\qquad
B=\operatorname{Var}\{\psi+C\varphi_\eta\}.
\]
\end{theorembox}
\begin{itemize}
  \item The sandwich variance includes \textbf{first-stage bridge estimation}.
  \item Individual-level bootstrap repeats both stages and label alignment.
  \item A growing sieve needs additional rate and inverse-problem conditions.
\end{itemize}
\end{frame}

\begin{frame}{Conclusion and limits}
\begin{columns}[T,onlytextwidth]
\begin{column}{.54\textwidth}
\begin{block}{Identification and estimation}
\begin{itemize}
  \item \textbf{Stage 1:} Missing IVs and complete-case completeness identify each supported-block law without a parametric selection model.
  \item \textbf{Stage 2:} Anchor items and $W$ identify latent-class proportions and all measurement kernels under a common label.
  \item \textbf{Estimator:} The bridge-weighted criterion follows the same two-stage order.

\end{itemize}

\begin{itemize}
  \item Class-proportion RMSE can exceed correctly specified selection likelihood.
  \item Identification assumes known $r$, anchor ordering, supported pairs, exclusion, completeness, local independence, and a finite latent-class model.
  \item Sieve inference requires additional inverse-problem theory.
\end{itemize}

\begin{frame}{Appendix: Proof of Theorem 1 (Steps 1--2)}
\begin{block}{Step 1: supported-pair laws}
New Proposition 1 identifies
\[
P(Y_a,Y_b,W,X),\qquad P(Y_j,Y_a,W,X),
\]
and therefore $\mathcal T_{abW}(x)$ and every $\mathcal T_{jaW}(x)$.
\end{block}
\begin{block}{Step 2: three-view latent-class representation}
Local independence gives
\[
\mathcal T_{abW}(x)
=\sum_{f=1}^r p_f(x)
M_a(:,f;x)\otimes M_b(:,f;x)\otimes G(:,f;x).
\]
\end{block}
\begin{takeaway}
$Y_a$, $Y_b$, and $W$ are the three conditionally independent views.
\end{takeaway}
\end{frame}

\begin{frame}{Appendix: Proof of Theorem 1 (Steps 3--4)}
\begin{block}{Step 3: Kruskal uniqueness}
With column Kruskal ranks $k_a,k_b,k_G$,
\[
\textcolor{Red}{\bm{k_a+k_b+k_G\ge 2r+2}}
\]
identifies $M_a,M_b,\widetilde G$ up to a common permutation and column scaling, where $\widetilde G(:,f)=p_fG(:,f)$.
\end{block}
\begin{block}{Step 4: scaling and labels}
Probability-column normalization gives
\[
p_f=\bm 1^\top\widetilde G(:,f),\qquad G(:,f)=\widetilde G(:,f)/p_f.
\]
The anchor ordering $\mu_{a1}<\cdots<\mu_{ar}$ fixes the common class labels.
\end{block}
\end{frame}

\begin{frame}{Appendix: Proof of Theorem 1 (Steps 5--6)}
\begin{block}{Step 5: rank for linear inversion}
The anchor Kruskal condition and Lemma 3.1 imply
\[
G\odot M_a\quad\text{has full column rank}.
\]
Hence the extension-tensor equation has a unique right-inverse solution.
\end{block}
\begin{block}{Step 6: latent class distribution given the shifter}
Because $p$, $G$, and $\Prb(W=w\mid X=x)$ are identified, Bayes' rule gives
\[
\lambda_f(w,x)
=\frac{p_f(x)G(w,f;x)}{\Prb(W=w\mid X=x)}.
\]
Thus $Q_F(x)=\{\lambda_f(w,x)\}_{f,w}$ is identified.
\end{block}
\end{frame}

\begin{frame}{Appendix: Proof of Theorem 1 (Steps 7--8)}
\begin{block}{Step 7: extension-kernel inversion}
For every nonanchor item $j$,
\[
M_j=T_{j\mid aW}
\left[\operatorname{diag}(p)(G\odot M_a)^\top\right]^+,
\]
so all measurement kernels $Q_Y=\{M_j\}_{j=1}^m$ are identified.
\end{block}
\begin{block}{Step 8: joint laws never observed together}
For any item set $A$,
\[
\Prb(Y_A=y_A\mid W=w,X=x)
=\sum_{f=1}^r\lambda_f(w,x)\prod_{j\in A}M_j(y_j,f;x).
\]
This includes sets $A$ that are never jointly observed.
\end{block}
\end{frame}

\begin{frame}{Appendix: empirical diagnostics and falsification checks}
For a proposed assignment of blockwise IVs, latent shifters, covariates, and anchor items, the data can diagnose:
\begin{itemize}
  \item relevance of each $Z_S$ and rank of a finite complete-case operator;
  \item supported-block positivity and effective sample size;
  \item third-mode rank, anchor separation, and label stability;
  \item agreement of measurement kernels recovered from redundant blocks.
\end{itemize}
\begin{takeaway}
Failure can refute a design, but success does not verify shadow exclusion, measurement exclusion, or local independence.
\end{takeaway}
\end{frame}
