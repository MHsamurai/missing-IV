\begin{frame}{Missing-data mechanisms and standard approaches}
\fontsize{9.2pt}{11pt}\selectfont
\begin{block}{Definition}
Little and Rubin (2002) classify missingness as Missing Completely At Random (MCAR), Missing At Random (MAR), and Nonignorable or Missing Not At Random (MNAR). Let $D$ denote the response pattern, and let $Y_{\mathrm{obs}}$ and $Y_{\mathrm{mis}}$ denote observed and missing data.
\end{block}
\vspace{.1em}
\setlength{\tabcolsep}{3pt}
\renewcommand{\arraystretch}{1.03}
\begin{tabular}{>{\raggedright\arraybackslash}p{.13\linewidth}
                >{\raggedright\arraybackslash}p{.45\linewidth}
                >{\raggedright\arraybackslash}p{.32\linewidth}}
\toprule
Mechanism & Assumption & Common approaches \\
\midrule
MCAR & Response does not depend on observed or missing data. $\Prb(D\mid Y)=\Prb(D)$ & Complete-case analysis, mean imputation \\
\addlinespace[.15em]
MAR & Response depends only on observed data. $\Prb(D\mid Y)=\Prb(D\mid Y_{\mathrm{obs}})$ & Multiple imputation, inverse probability weighting \\
\addlinespace[.15em]
NI/MNAR & Response also depends on missing data. $\Prb(D\mid Y)=\Prb(D\mid Y_{\mathrm{obs}},Y_{\mathrm{mis}})$ & Selection models, pattern-mixture models, \textcolor{Red}{\bfseries missingness IV} \\
\bottomrule
\end{tabular}
\vspace{.35em}
\begin{takeaway}
This paper studies MNAR item nonresponse and uses missingness IVs to recover full-data laws before identifying the latent structure.
\end{takeaway}
\end{frame}

\begin{frame}{Missingness IV}
Let $R$ indicate that the outcome $Y$ is observed, and let $Z$ be an always-observed variable associated with $Y$.
\begin{columns}[T,onlytextwidth]
\begin{column}{.47\textwidth}
\begin{block}{Usual IV}
A usual instrumental variable affects selection but has no direct effect on the outcome. By itself, it generally does not point identify the full distribution of $Y$.
\end{block}
\end{column}
\begin{column}{.50\textwidth}
\begin{assumptionbox}{Missingness IV / shadow variable}
\[
R\indep Z\mid Y,
\]
while $Z$ remains associated with $Y$.
\end{assumptionbox}
\end{column}
\end{columns}
Under positivity, this exclusion yields the observable conditional moment equation
\[
\E\left\{\frac{R}{\pi(Y)}\middle|Z\right\}=1,
\qquad \pi(y)=\Prb(R=1\mid Y=y).
\]
Completeness of $Y$ with respect to $Z$ makes its solution unique (d'Haultfoeuille, 2010).
\end{frame}

\begin{frame}{A simple multivariate extension}
Let $D=(D_1,\ldots,D_m)'$, $Y=(Y_1,\ldots,Y_m)'$, and $Z$ denote the response indicators, multivariate outcomes, and missingness IVs. Define the \term{global complete-case indicator}
\[
R=\mathbf 1\{D=\mathbf 1\}=\prod_{j=1}^m D_j.
\]
If
\[
R\indep Z\mid Y,\qquad \Prb(R=1\mid Y)>0,
\]
and $Y$ is complete with respect to $Z$, then
\[
\E\left\{\frac{R}{\pi(Y)}\middle|Z\right\}=1
\]
uniquely identifies $\pi(y)=\Prb(R=1\mid Y=y)$ and hence the full-data law of $Y$.
\begin{takeaway}
This direct extension is valid, but it relies on observing the entire vector $Y$ in a global complete case.
\end{takeaway}
\end{frame}

\begin{frame}{Latent measurement model}
\begin{columns}[c,onlytextwidth]
\begin{column}{.48\textwidth}
\centering
\resizebox{.98\linewidth}{!}{\input{notes/figures/latent_measurement_model.tex}}
\end{column}
\begin{column}{.49\textwidth}
Let $F$ denote an \term{unobserved latent variable} and let $Y=(Y_1,\ldots,Y_m)$ denote its potential \term{observed measurements}.
\begin{itemize}
  \item $Q_F(\cdot\mid W,X)$ is the latent distribution.
  \item $Q_j(\cdot\mid F,X)$ is the measurement kernel for item $j$.
  \item $Y_1,\ldots,Y_m$ are conditionally independent given $(F,X)$.
\end{itemize}
\end{column}
\end{columns}
\vspace{.2em}
\[
\Prb(Y=y\mid W=w,X=x)
=\int\prod_{j=1}^m Q_j(y_j\mid f,x)\,dQ_F(f\mid w,x).
\]
\begin{takeaway}
Identifying the left-hand side is insufficient: \redkey{$Q_F$ and $Q_Y$ require a unique tensor decomposition.}
\end{takeaway}
\end{frame}

\begin{frame}{A latent shifter is not a missingness IV}
Suppose $D\indep W\mid(Y,F,X)$. After integrating out $F$,
\[
\Prb(D=d\mid Y=y,W=w,X=x)
=\int \Prb(D=d\mid y,f,x)\,d\Prb(f\mid y,w,x).
\]
If $W$ shifts $F$, the mixing distribution generally depends on $w$. Hence
\[
D\indep W\mid(Y,X)\quad\text{does }\textcolor{Red}{\mathbf{not}}\text{ follow.}
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{$Z_S$: missingness IV}
Predicts $Y_S$ and satisfies an exclusion for $R_S$ conditional on $(Y_S,U_S)$.
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\begin{block}{$W$: latent shifter}
Moves the distribution of $F$ and supplies third-mode variation for decomposition.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Observed data and supported blocks}
Let $D_j$ denote the response indicator for $Y_j$. The observed-data vector is
\[
O=(X,W,Z,D,Y_D),\qquad Y_D=\{Y_j:D_j=1\}.
\]
For a block $S\subseteq\{1,\ldots,m\}$, define
\[
R_S=\prod_{j\in S}D_j,\qquad Y_S=(Y_j:j\in S),\qquad U_S=(X,W).
\]
$Y_S$ is observed only when $R_S=1$. A block is \term{supported} if
\[
\pi_S(y,u):=\Prb(R_S=1\mid Y_S=y,U_S=u)>0
\]
on the target support.
\begin{takeaway}
The method requires supported anchor and extension pairs, not a global complete case.
\end{takeaway}
\end{frame}

\begin{frame}{Stage-1 assumptions}
\begin{assumptionbox}{Blockwise covariate-adjusted missingness IV}
For every supported block $S$,
\[
R_S\indep Z_S\mid(Y_S,U_S).
\]
\end{assumptionbox}
\begin{assumptionbox}{Complete-case completeness}
For $h\in L^2\{\Prb(Y_S,U_S\mid R_S=1)\}$,
\[
\E\{h(Y_S,U_S)\mid R_S=1,Z_S,U_S\}=0
\Longrightarrow h(Y_S,U_S)=0.
\]
\end{assumptionbox}
Positivity transfers null sets between the complete-case and full laws; completeness makes the observed conditional operator injective.
\end{frame}

\begin{frame}{Proposition 1: identification of a supported-block law}
\begin{theorembox}{Supported-block identification}
Under blockwise exclusion, positivity, and complete-case completeness, $\pi_S$ is the unique positive solution of
\[
\E\left\{\frac{R_S}{q(Y_S,U_S)}\middle|Z_S,U_S\right\}=1.
\]
Consequently, for every integrable $h$,
\[
\E h(Y_S,Z_S,U_S)
=\E\left\{\frac{R_Sh(Y_S,Z_S,U_S)}{\pi_S(Y_S,U_S)}\right\},
\]
and $\Prb(Y_S,Z_S,U_S)$ is identified.
\end{theorembox}
\begin{takeaway}
Each block law is recovered without specifying a parametric selection model.
\end{takeaway}
\end{frame}

\begin{frame}{Proof sketch: the observable bridge is unique}
\begin{enumerate}
  \item Exclusion implies $\E(R_S\mid Y_S,Z_S,U_S)=\pi_S(Y_S,U_S)$, so the true $\pi_S$ satisfies the bridge moment.
  \item Any other positive solution $q$ satisfies
  \[
  \E\{q^{-1}-\pi_S^{-1}\mid R_S=1,Z_S,U_S\}=0.
  \]
  \item Complete-case completeness gives $q^{-1}=\pi_S^{-1}$ on $R_S=1$.
  \item Positivity extends equality to the full support; inverse weighting recovers every block functional.
\end{enumerate}
\begin{takeaway}
The proof separates \bluekey{bridge uniqueness} from \bluekey{latent decomposition}.
\end{takeaway}
\end{frame}

\begin{frame}{Finite latent classes and the anchor tensor}
Fix $X=x$ and let $F\in\{1,\ldots,r\}$. Define
\[
M_j(y,f;x)=\Prb(Y_j=y\mid F=f,X=x),\quad
p_f(x)=\Prb(F=f\mid X=x),
\]
\[
G(w,f;x)=\Prb(W=w\mid F=f,X=x).
\]
The recovered anchor law forms a three-way array:
\[
\mathcal T_{abW}(x)
=\sum_{f=1}^r p_f(x)
M_a(:,f;x)\otimes M_b(:,f;x)\otimes G(:,f;x).
\]
\begin{takeaway}
$Y_a$, $Y_b$, and $W$ act as the three conditionally independent views.
\end{takeaway}
\end{frame}

\begin{frame}{Anchor-pair identification conditions}
\begin{assumptionbox}{Covariate-assisted anchor-pair structure}
For every $x$ in a common full-probability set:
\begin{itemize}
  \item the anchor pair $\{a,b\}$ and all extension pairs $\{j,a\}$ are supported;
  \item $p_f(x)>0$ and $\Prb(W=w\mid X=x)>0$;
  \item with column Kruskal ranks $k_a,k_b,k_G$,
  \[
  \textcolor{Red}{\bm{k_a+k_b+k_G\ge 2r+2}};
  \]
  \item a known anchor score satisfies $\mu_{a1}(x)<\cdots<\mu_{ar}(x)$ and fixes the common labels.
\end{itemize}
\end{assumptionbox}
The rank condition identifies the anchor decomposition; overlapping pairs propagate it to every $M_j$.
\end{frame}

\begin{frame}{Theorem 1: integrated identification}
\begin{theorembox}{Anchor-pair identification with $W$ as the third mode}
Under the stage-1 assumptions and the anchor-pair conditions, the observed-data law identifies, under the fixed anchor ordering,
\[
p(x),\quad G(x),\quad
Q_F(x)=\{\lambda_f(w,x)\}_{f,w},\quad
Q_Y(x)=\{M_j(y,f;x)\}_{j,y,f}.
\]
Moreover, for any set of items $A$,
\[
\Prb(Y_A=y_A\mid W=w,X=x)
=\sum_{f=1}^r\lambda_f(w,x)\prod_{j\in A}M_j(y_j,f;x)
\]
is identified even if all items in $A$ are never jointly observed.
\end{theorembox}
\end{frame}

\begin{frame}{Proof sketch: from the anchor tensor to all kernels}
\begin{enumerate}
  \item \bluekey{Recover tensors}: Proposition 1 identifies $\mathcal T_{abW}$ and each $\mathcal T_{jaW}$.
  \item \bluekey{Decompose the anchor}: Kruskal uniqueness identifies $M_a,M_b,p,G$; simplex normalization and anchor ordering remove scaling and permutation.
  \item \bluekey{Recover class weights}: Bayes' rule gives
  \[
  \lambda_f(w,x)=\frac{p_f(x)G(w,f;x)}{\Prb(W=w\mid X=x)}.
  \]
  \item \bluekey{Extend to every item}: since $G\odot M_a$ has full column rank,
  \[
  M_j=T_{j\mid aW}\left[\operatorname{diag}(p)(G\odot M_a)^\top\right]^+.
  \]
\end{enumerate}
\end{frame}

\begin{frame}{Bridge-weighted composite estimation}
Let $\mathcal S_*=\{\{a,b\}\}\cup\{\{j,a\}:j\notin\{a,b\}\}$. Following the identification order,
\[
\widehat\theta=\arg\max_{\theta\in\Theta}
\frac1n\sum_{i=1}^n\sum_{S\in\mathcal S_*}c_S
\frac{R_{iS}}{\widehat\pi_S(Y_{iS},U_i)}
\log p_\theta(Y_{iS},W_i\mid X_i),
\]
where $\theta=\{p,G,M_1,\ldots,M_m\}$.
\begin{itemize}
  \item This is a \term{composite M-estimator}, not a full likelihood.
  \item The bridge may be finite-dimensional or a regularized sieve.
  \item Inference must include first-stage bridge estimation.
\end{itemize}
\begin{theorembox}{Finite-dimensional bridge}
Under regularity and separation, $\widehat\theta\to_p\theta_0$ and
$\sqrt n(\widehat\theta-\theta_0)\rightsquigarrow N(0,A^{-1}BA^{-\top})$.
\end{theorembox}
\end{frame}

\begin{frame}{Computation}
\begin{enumerate}
  \item Estimate the bridge for every supported pair and check overlap.
  \item Construct weighted anchor and extension tensors.
  \item Initialize a nonnegative CP decomposition of the anchor tensor.
  \item Normalize columns, fix labels, and invert extension tensors.
  \item Run constrained block-coordinate updates from multiple starts.
  \item Select the largest criterion and report rank, anchor, and propensity gaps.
\end{enumerate}
\begin{takeaway}
The algorithm follows the same order as the identification argument.
\end{takeaway}
\end{frame}

\begin{frame}{Interpretation and limits}
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{Advantage}
The bridge avoids bias from specifying the wrong parametric selection equation while recovering both measurement kernels and class proportions.
\end{block}
\end{column}
\begin{column}{.49\textwidth}
\begin{block}{Cost and limits}
\begin{itemize}
  \item Class-proportion RMSE remains above correctly specified selection likelihood.
  \item Robustness does not cover violations of exclusion, completeness, local independence, or the latent-class model.
  \item Sieve inference requires additional inverse-problem theory.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\vspace{.35em}
\begin{takeaway}
The gain is robustness to the \redkey{selection-link specification}, not universal robustness or efficiency dominance.
\end{takeaway}
\end{frame}

\begin{frame}{Conclusion}
\begin{enumerate}
  \item Shadow IVs and complete-case completeness identify each supported-block law without a parametric selection model.
  \item Anchor items connect overlapping blocks, while $W$ supplies the third mode needed for tensor uniqueness.
  \item The observed-data law then identifies finite latent-class proportions and all measurement kernels under a common label.
  \item A bridge-weighted composite estimator follows the same order and performs well under selection-model misspecification.
\end{enumerate}
\vspace{.55em}
\begin{takeaway}
\bluekey{Recover block laws first; decompose the latent structure second.}
\end{takeaway}
\end{frame}

For $j\notin\{a,b\}$, mode-$j$ unfolding gives
\[
T_{j\mid aW}(x)
=M_j(x)\operatorname{diag}\{p(x)\}
\{G(x)\odot M_a(x)\}^{\top}.
\]
The anchor Kruskal condition implies that $G\odot M_a$ has full column rank. Thus
\[
M_j(x)=T_{j\mid aW}(x)
\left[\operatorname{diag}\{p(x)\}
\{G(x)\odot M_a(x)\}^{\top}\right]^+.
\]
The same anchor ordering fixes the label of every extension kernel.
\end{frame}

\begin{frame}{Appendix: empirical diagnostics}
For a proposed assignment of blockwise IVs, latent shifters, covariates, and anchor items, the data can diagnose:
\begin{itemize}
  \item relevance of each $Z_S$ and rank of a finite complete-case operator;
  \item supported-block positivity and effective sample size;
  \item third-mode rank, anchor separation, and label stability;
  \item agreement of measurement kernels recovered from redundant blocks.
\end{itemize}
\begin{takeaway}
Failure can refute a design, but success does not verify shadow exclusion, measurement exclusion, or local independence.
\end{takeaway}
\end{frame}
