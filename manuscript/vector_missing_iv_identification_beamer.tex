\begin{frame}{Missing-data mechanisms and standard approaches}
\fontsize{9.2pt}{11pt}\selectfont
\begin{block}{Definition}
Little and Rubin (2002) classify missingness as MCAR, MAR, and NI/MNAR. Let $D$ denote the response pattern, and let $Y_{\mathrm{obs}}$ and $Y_{\mathrm{mis}}$ denote observed and missing data. The auxiliary variable used below is a \textcolor{Red}{\bfseries Missing IV (also called a shadow variable)}.
\end{block}
\vspace{.1em}
\setlength{\tabcolsep}{3pt}
\renewcommand{\arraystretch}{1.03}
\begin{tabular}{>{\raggedright\arraybackslash}p{.13\linewidth}
                >{\raggedright\arraybackslash}p{.45\linewidth}
                >{\raggedright\arraybackslash}p{.32\linewidth}}
\toprule
Mechanism & Assumption & Common approaches \\
\midrule
MCAR & Response does not depend on observed or missing data. $\Prb(D\mid Y)=\Prb(D)$ & Complete-case analysis, mean imputation \\
\addlinespace[.15em]
MAR & Response depends only on observed data. $\Prb(D\mid Y)=\Prb(D\mid Y_{\mathrm{obs}})$ & Multiple imputation, inverse probability weighting \\
\addlinespace[.15em]
NI/MNAR & Response also depends on missing data. $\Prb(D\mid Y)=\Prb(D\mid Y_{\mathrm{obs}},Y_{\mathrm{mis}})$ & Selection models, pattern-mixture models, \textcolor{Red}{\bfseries Missing IV} \\
\bottomrule
\end{tabular}
\vspace{.35em}
\begin{takeaway}
Under MNAR, respondent-only analysis can induce sample-selection bias (Heckman, 1979). Missing IVs recover full-data laws before latent decomposition.
\end{takeaway}
\end{frame}

Let $R$ indicate that $Y$ is observed and let $Z$ be an always-observed auxiliary variable.

$Z$ predicts $Y$ and satisfies $R\indep Z\mid Y$.
\end{block}
\end{column}
\end{columns}
\vspace{.15em}
For a Missing IV, positivity gives
\[
\E\left\{\frac{R}{\pi(Y)}\middle|Z\right\}=1,
\qquad \pi(y)=\Prb(R=1\mid Y=y),
\]
and completeness makes $\pi$ unique (d'Haultfoeuille, 2010, \emph{JoE}).
\end{frame}

\begin{frame}{A simple multivariate extension}
Let $D=(D_1,\ldots,D_m)'$, $Y=(Y_1,\ldots,Y_m)'$, and $Z$ denote the response indicators, multivariate outcomes, and missing IVs. Define the \term{global complete-case indicator}
\[
R=\mathbf 1\{D=\mathbf 1\}=\prod_{j=1}^m D_j.
\]
If
\[
R\indep Z\mid Y,\qquad \Prb(R=1\mid Y)>0,
\]
and $Y$ is complete with respect to $Z$, then
\[
\E\left\{\frac{R}{\pi(Y)}\middle|Z\right\}=1
\]
uniquely identifies $\pi(y)=\Prb(R=1\mid Y=y)$ and hence the full-data law of $Y$.
\begin{takeaway}
This direct extension is valid, but it relies on observing the entire vector $Y$ in a global complete case.
\end{takeaway}
\end{frame}

\begin{frame}{Latent measurement model}
\begin{columns}[c,onlytextwidth]
\begin{column}{.48\textwidth}
\centering
\resizebox{.98\linewidth}{!}{\input{notes/figures/latent_measurement_model.tex}}
\end{column}
\begin{column}{.49\textwidth}
Let $F$ denote an \term{unobserved latent variable} and let $Y=(Y_1,\ldots,Y_m)$ denote its potential \term{observed measurements}.
\begin{itemize}
  \item $Q_F(\cdot\mid W,X)$ is the latent distribution.
  \item $Q_j(\cdot\mid F,X)$ is the measurement kernel for item $j$.
  \item $Y_1,\ldots,Y_m$ are conditionally independent given $(F,X)$.
\end{itemize}
\end{column}
\end{columns}
\vspace{.2em}
\[
\Prb(Y=y\mid W=w,X=x)
=\int\prod_{j=1}^m Q_j(y_j\mid f,x)\,dQ_F(f\mid w,x).
\]
\begin{takeaway}
Identifying the left-hand side is insufficient: \redkey{$Q_F$ and $Q_Y$ require a unique tensor decomposition.}
\end{takeaway}
\end{frame}

\begin{frame}{A latent shifter is not a Missing IV}
Suppose $D\indep W\mid(Y,F,X)$. After integrating out $F$,
\[
\Prb(D=d\mid Y=y,W=w,X=x)
=\int \Prb(D=d\mid y,f,x)\,d\Prb(f\mid y,w,x).
\]
If $W$ shifts $F$, the mixing distribution generally depends on $w$. Hence
\[
D\indep W\mid(Y,X)\quad\text{does }\textcolor{Red}{\mathbf{not}}\text{ follow.}
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{$Z_S$: Missing IV}
Predicts $Y_S$ and satisfies an exclusion for $R_S$ conditional on $(Y_S,U_S)$.
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\begin{block}{$W$: latent shifter}
Moves the distribution of $F$ and supplies third-mode variation for decomposition.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Observed data and supported blocks}
\begin{block}{Supported block}
A subset $S\subseteq\{1,\ldots,m\}$ is \term{supported} when all measurements in $Y_S$ are jointly observed with positive probability on the target support. A global complete case is not required.
\end{block}
\begin{columns}[c,onlytextwidth]
\begin{column}{.53\textwidth}
Let $D_j$ indicate response for $Y_j$. Then
\[
O=(X,W,Z,D,Y_D),\quad Y_D=\{Y_j:D_j=1\},
\]
\[
R_S=\prod_{j\in S}D_j,\quad
Y_S=(Y_j:j\in S),\quad U_S=(X,W).
\]
The support condition is
\[
\pi_S(y,u):=\Prb(R_S=1\mid Y_S=y,U_S=u)>0.
\]
\end{column}

\begin{frame}{Stage-1 assumptions}
\begin{block}{Assumption 6: Blockwise covariate-adjusted Missing IV}
For every supported block $S$,
\[
R_S\indep Z_S\mid(Y_S,U_S).
\]
\end{block}
\begin{block}{Assumption 7: Complete-case completeness}
For $h\in L^2\{\Prb(Y_S,U_S\mid R_S=1)\}$,
\[
\E\{h(Y_S,U_S)\mid R_S=1,Z_S,U_S\}=0
\Longrightarrow h(Y_S,U_S)=0.
\]
\end{block}
Positivity transfers null sets between the complete-case and full laws; completeness makes the observed conditional operator injective.
\end{frame}

\begin{frame}{\textcolor{Red}{New!} Proposition 1}
\begin{theorembox}{Supported-block identification}
Under blockwise exclusion, positivity, and complete-case completeness, $\pi_S$ is the unique positive solution of
\[
\E\left\{\frac{R_S}{q(Y_S,U_S)}\middle|Z_S,U_S\right\}=1.
\]
Consequently, for every integrable $h$,
\[
\E h(Y_S,Z_S,U_S)
=\E\left\{\frac{R_Sh(Y_S,Z_S,U_S)}{\pi_S(Y_S,U_S)}\right\},
\]
and $\Prb(Y_S,Z_S,U_S)$ is identified.
\end{theorembox}
\begin{takeaway}
Each block law is recovered without specifying a parametric selection model.
\end{takeaway}
\end{frame}

\begin{block}{After Stage 1}
The full law of every required $Y_S$ is identified, but the latent variable $F$ remains unobserved.

Fix $X=x$ and let $F\in\{1,\ldots,r\}$. Define
\[
\begin{aligned}

\begin{frame}{Stage-2 identification conditions}
\begin{assumptionbox}{Covariate-assisted anchor-pair structure}
For every $x$ in a common full-probability set:
\begin{itemize}
  \item the anchor pair $\{a,b\}$ and all extension pairs $\{j,a\}$ are supported;
  \item $p_f(x)>0$ and $\Prb(W=w\mid X=x)>0$;
  \item with column Kruskal ranks $k_a,k_b,k_G$,
  \[
  \textcolor{Red}{\bm{k_a+k_b+k_G\ge 2r+2}};
  \]
  \item a known anchor score satisfies $\mu_{a1}(x)<\cdots<\mu_{ar}(x)$ and fixes the common labels.
\end{itemize}
\bluekey{The three-view Kruskal condition is the same as Allman et al. (2009, Theorem 1).}
\end{assumptionbox}
Supported overlapping pairs and anchor ordering adapt that condition to the present MNAR problem.
\end{frame}

\begin{frame}{Theorem 1}
\begin{theorembox}{Anchor-pair identification with $W$ as the third mode}
Under the stage-1 assumptions and the anchor-pair conditions, the observed-data law identifies, under the fixed anchor ordering,
\[
p(x),\quad G(x),\quad
Q_F(x)=\{\lambda_f(w,x)\}_{f,w},\quad
Q_Y(x)=\{M_j(y,f;x)\}_{j,y,f}.
\]
Moreover, for any set of items $A$,
\[
\Prb(Y_A=y_A\mid W=w,X=x)
=\sum_{f=1}^r\lambda_f(w,x)\prod_{j\in A}M_j(y_j,f;x)
\]
is identified even if all items in $A$ are never jointly observed.
\end{theorembox}
\end{frame}

\begin{frame}{Steps 1--2: recover pair laws and form the anchor tensor}
\begin{block}{Step 1: supported-pair laws}
New Proposition 1 identifies
\[
P(Y_a,Y_b,W,X),\qquad P(Y_j,Y_a,W,X),
\]
and therefore $\mathcal T_{abW}(x)$ and every $\mathcal T_{jaW}(x)$.
\end{block}
\begin{block}{Step 2: three-view latent-class representation}
Local independence gives
\[
\mathcal T_{abW}(x)
=\sum_{f=1}^r p_f(x)
M_a(:,f;x)\otimes M_b(:,f;x)\otimes G(:,f;x).
\]
\end{block}
\begin{takeaway}
$Y_a$, $Y_b$, and $W$ are the three conditionally independent views.
\end{takeaway}
\end{frame}

\begin{frame}{Steps 3--4: identify and normalize the anchor decomposition}
\begin{block}{Step 3: Kruskal uniqueness}
With column Kruskal ranks $k_a,k_b,k_G$,
\[
\textcolor{Red}{\bm{k_a+k_b+k_G\ge 2r+2}}
\]
identifies $M_a,M_b,\widetilde G$ up to a common permutation and column scaling, where $\widetilde G(:,f)=p_fG(:,f)$.
\end{block}
\begin{block}{Step 4: scaling and labels}
Probability-column normalization gives
\[
p_f=\bm 1^\top\widetilde G(:,f),\qquad G(:,f)=\widetilde G(:,f)/p_f.
\]
The anchor ordering $\mu_{a1}<\cdots<\mu_{ar}$ fixes the common class labels.
\end{block}
\end{frame}

\begin{frame}{Steps 5--6: obtain inversion rank and class weights}
\begin{block}{Step 5: rank for linear inversion}
The anchor Kruskal condition and the Khatri--Rao rank bound imply
\[
G\odot M_a\quad\text{has full column rank}.
\]
Hence the extension-tensor equation has a unique right-inverse solution.
\end{block}
\begin{block}{Step 6: latent class distribution given the shifter}
Because $p$, $G$, and $\Prb(W=w\mid X=x)$ are identified, Bayes' rule gives
\[
\lambda_f(w,x)
=\frac{p_f(x)G(w,f;x)}{\Prb(W=w\mid X=x)}.
\]
Thus $Q_F(x)=\{\lambda_f(w,x)\}_{f,w}$ is identified.
\end{block}
\end{frame}

\begin{frame}{Steps 7--8: recover all kernels and unobserved joint laws}
\begin{block}{Step 7: extension-kernel inversion}
For every nonanchor item $j$,
\[
M_j=T_{j\mid aW}
\left[\operatorname{diag}(p)(G\odot M_a)^\top\right]^+,
\]
so all measurement kernels $Q_Y=\{M_j\}_{j=1}^m$ are identified.
\end{block}
\begin{block}{Step 8: joint laws never observed together}
For any item set $A$,
\[
\Prb(Y_A=y_A\mid W=w,X=x)
=\sum_{f=1}^r\lambda_f(w,x)\prod_{j\in A}M_j(y_j,f;x).
\]
This includes sets $A$ that are never jointly observed.
\end{block}
\end{frame}

\begin{frame}{Bridge-weighted composite estimation}
Let $\mathcal S_*=\{\{a,b\}\}\cup\{\{j,a\}:j\notin\{a,b\}\}$. Following the identification order,
\[
\widehat\theta=\arg\max_{\theta\in\Theta}
\frac1n\sum_{i=1}^n\sum_{S\in\mathcal S_*}c_S
\frac{R_{iS}}{\widehat\pi_S(Y_{iS},U_i)}
\log p_\theta(Y_{iS},W_i\mid X_i),
\]
where $\theta=\{p,G,M_1,\ldots,M_m\}$.
\begin{itemize}
  \item This is a \term{composite M-estimator}, not a full likelihood.
  \item The bridge may be finite-dimensional or a regularized sieve.
  \item Inference must include first-stage bridge estimation.
\end{itemize}
\begin{theorembox}{Finite-dimensional bridge}
Under regularity and separation, $\widehat\theta\to_p\theta_0$ and
$\sqrt n(\widehat\theta-\theta_0)\rightsquigarrow N(0,A^{-1}BA^{-\top})$.
\end{theorembox}
\end{frame}

\begin{frame}{Conclusion and limits}
\begin{columns}[T,onlytextwidth]
\begin{column}{.54\textwidth}
\begin{block}{Identification and estimation}
\begin{itemize}
  \item \textbf{Stage 1:} Missing IVs and complete-case completeness identify each supported-block law without a parametric selection model.
  \item \textbf{Stage 2:} Anchor items and $W$ identify latent-class proportions and all measurement kernels under a common label.
  \item \textbf{Estimator:} The bridge-weighted criterion follows the same two-stage order.
  \item \textbf{Simulation:} Bias and RMSE improve relative to MAR and misspecified selection likelihood.
\end{itemize}
\end{block}
\end{column}
\begin{column}{.43\textwidth}
\begin{block}{Limits}
\begin{itemize}
  \item Class-proportion RMSE can exceed correctly specified selection likelihood.
  \item Robustness requires exclusion, completeness, local independence, and the latent-class model.
  \item Sieve inference requires additional inverse-problem theory.
\end{itemize}
\end{block}
\vspace{.3em}
\redkey{Robust to selection-link misspecification, not universally robust or efficiency-dominant.}
\end{column}
\end{columns}

\begin{frame}{Appendix: extension-kernel inversion}
For $j\notin\{a,b\}$, mode-$j$ unfolding gives
\[
T_{j\mid aW}(x)
=M_j(x)\operatorname{diag}\{p(x)\}
\{G(x)\odot M_a(x)\}^{\top}.
\]
The anchor Kruskal condition implies that $G\odot M_a$ has full column rank. Thus
\[
M_j(x)=T_{j\mid aW}(x)
\left[\operatorname{diag}\{p(x)\}
\{G(x)\odot M_a(x)\}^{\top}\right]^+.
\]
The same anchor ordering fixes the label of every extension kernel.
\end{frame}

\begin{frame}{Appendix: empirical diagnostics}
For a proposed assignment of blockwise IVs, latent shifters, covariates, and anchor items, the data can diagnose:
\begin{itemize}
  \item relevance of each $Z_S$ and rank of a finite complete-case operator;
  \item supported-block positivity and effective sample size;
  \item third-mode rank, anchor separation, and label stability;
  \item agreement of measurement kernels recovered from redundant blocks.
\end{itemize}
\begin{takeaway}
Failure can refute a design, but success does not verify shadow exclusion, measurement exclusion, or local independence.
\end{takeaway}
\end{frame}
