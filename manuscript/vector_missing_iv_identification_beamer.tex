\begin{frame}{Missing-data mechanisms and standard approaches}
Let $D$ be the response pattern; $Y=(Y_{\mathrm{obs}},Y_{\mathrm{mis}})$.
\smallskip
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{@{}p{.19\linewidth}p{.40\linewidth}p{.36\linewidth}@{}}
\toprule
Mechanism & Response probability & Common approaches\\
\midrule
MCAR & $\Prb(D\mid Y)=\Prb(D)$ & Complete-case analysis\\[.5em]
MAR & $\Prb(D\mid Y)=\Prb(D\mid Y_{\mathrm{obs}})$ & Multiple imputation; IPW\\[.5em]
MNAR & May depend on $Y_{\mathrm{mis}}$ even given $Y_{\mathrm{obs}}$ & Selection / pattern-mixture models; \redkey{Missing IV}\\
\bottomrule
\end{tabular}
\medskip
MCAR: missing completely at random; MAR: missing at random; MNAR: missing not at random.\vspace{-1pt}
\begin{takeaway}
Under MNAR, respondent-only analysis can induce selection bias. A \redkey{Missing IV / shadow variable} can identify the full-data law under suitable assumptions.
\end{takeaway}
{\slideannotation Little and Rubin (2002); Heckman (1979, \emph{Econometrica}).}
\end{frame}

Let $R=1$ indicate that $Y$ is observed; $Z$ is always observed.

\end{block}

\begin{column}{.50\textwidth}
\begin{block}{Missing IV}
\centering
\begin{tikzpicture}[>=Latex,node distance=1.05cm,
  var/.style={circle,draw=Navy,fill=white,minimum size=.78cm,thick},

  \draw[path] (z) -- (y);

\end{tikzpicture}
\par
$Z$ predicts $Y$ and satisfies $R\indep Z\mid Y$.
\end{block}
\end{column}

Missing-IV exclusion + positivity give the moment; completeness identifies $\pi$:
\setlength{\abovedisplayskip}{2pt}
\setlength{\belowdisplayskip}{0pt}
\[
\E\left\{\frac{R}{\pi(Y)}\middle|Z\right\}=1,
\qquad \pi(y)=\Prb(R=1\mid Y=y),
\]
{\footnotesize
\begin{itemize}
\setlength{\itemsep}{0pt}
\item d'Haultfoeuille (2010, \emph{JoE}): full law via conditional moments.
\item Zhao and Shao (2015, \emph{JASA}): covariate-adjusted GLM identification.
\item Miao et al. (2019, arXiv): shadow-variable full-law identification using complete-case completeness.
\end{itemize}
}
\par
\bluekey{IPW recovers the full-data law; its latent components remain to be identified.}
\end{frame}

\begin{frame}{From outcome-law recovery to latent identification}
\bluekey{Recovering the outcome law does not by itself identify its latent components.}
\par\smallskip
Our illustrative representation: latent $F$, observed covariates $U$ (conditioned on).
\par\smallskip
{\centering
\begin{tikzpicture}[>=Latex,node distance=1.5cm,
 observed/.style={rectangle,draw=Navy,minimum size=.7cm,thick},

\draw[path] (z)--(f);

\[
\bigl\{P(F\mid Z,U),P(Y\mid F,U)\bigr\}
\ \xrightarrow{\text{integrate out }F}\ P(Y\mid Z,U).
\]
\begin{block}{Two different identification targets}
\begin{itemize}
\item \textbf{Missing IV:} observed-data law $\longrightarrow$ outcome law
(or GLM parameters under Zhao--Shao's model).
\item \textbf{Our question:} recovered outcome laws $\longrightarrow$
\redkey{latent distribution and measurement kernels?}
\end{itemize}
\end{block}
\bluekey{We study conditions for this additional decomposition.}
\par\smallskip
{\slideannotation d'Haultfoeuille (2010, \emph{JoE}); Zhao and Shao (2015, \emph{JASA}). The DAG is our illustration.}
\end{frame}

\begin{frame}{Latent measurement model}
\begin{columns}[c,onlytextwidth]
\begin{column}{.48\textwidth}
\centering
\input{manuscript/figures/readable/latent_measurement_model.tex}
\end{column}
\begin{column}{.49\textwidth}
In structural equation modeling (SEM) and item response theory (IRT), \term{latent} $F$ underlies measurements $Y=(Y_1,\ldots,Y_m)$.
\begin{itemize}
  \item $Q_F(\cdot\mid W,X)$ is the latent distribution.
  \item $Q_j(\cdot\mid F,X)$ is the item measurement kernel.
  \item Items are conditionally independent given $(F,X)$.
\end{itemize}
\end{column}
\end{columns}
\vspace{.2em}
\[
\Prb(Y=y\mid W=w,X=x)
=\int\prod_{j=1}^m Q_j(y_j\mid f,x)\,dQ_F(f\mid w,x).
\]
{\footnotesize
Allman et al. (2009, \emph{AoS}) \textbf{identify finite class proportions and measurement kernels} up to labels under three-view rank conditions.\par
}
\end{frame}

\begin{frame}{Latent models with nonignorable missingness}
\begin{itemize}
\item
\textbf{Latent models and inference under nonignorable missingness} have long been studied in biometrics and psychometrics (Muth\'en et al., 1987, \emph{Psychometrika}; Holman and Glas, 2005, \emph{BJMSP}; Lee and Tang, 2006, \emph{Psychometrika}; Harel and Schafer, 2009, \emph{Biometrika}; Jung et al., 2011, \emph{CSDA}; Kuha et al., 2018, \emph{JRSS A}).\par
\item Kano and Takai (2011, \emph{JMVA}) \textbf{consistently estimate linear SEMs} under identification and latent ignorability.
\item Xie et al. (2026, arXiv) study MNAR identification with deep latent models.
\end{itemize}
\end{frame}

\begin{frame}{Why leave the selection model unspecified?}
\par{\footnotesize Zhao and Shao (2015, \emph{JASA}, Sections 1 and 6) motivate semiparametric estimation.}
\begin{itemize}
  \item \textbf{Full parametric likelihood:} can be efficient if correct; selection-link errors can induce bias.
  \item \textbf{Semiparametric route:} outcome model specified; selection mechanism unspecified.
  \item \textbf{Nonparametric recovery:} no selection link; requires Missing IV and completeness.
\end{itemize}
\begin{takeaway}
\bluekey{Recovering the law of $Y$ does not itself separate $Q_F$ and $Q_Y$.}
\end{takeaway}
\end{frame}

\begin{frame}{Stage 1 (1): Recover the block law}
\bluekey{Recover the population law, including people with missing items.}\par
\smallskip
Let $Y_S$ be the items in block $S$, $R_S=1$ indicate that all are observed,
and $U_S=(X,W)$ be observed covariates.
\begin{block}{\textcolor{Red}{New!} Proposition 1: Key result}
With Missing IV $Z_S$, positivity and complete-case completeness,
\[
\E\left\{\frac{R_S}{\pi_S(Y_S,U_S)}\middle|Z_S,U_S\right\}=1,
\qquad \pi_S(y,u)=\Prb(R_S=1\mid Y_S=y,U_S=u).
\]
The moment identifies $\pi_S$ in the stated bridge class, hence the
block law $\Prb(Y_S,Z_S,U_S)$ is identified.
\end{block}
\begin{itemize}
\item Weight observed blocks by $1/\pi_S$ to correct MNAR selection.
\item \redkey{No parametric selection model}; exclusion and completeness are required.
\end{itemize}
{\slideannotation Assumptions 1, 5--7 and $\pi_S^{-1}\in L^2\{P(Y_S,U_S\mid R_S=1)\}$; details in Appendix.}
\end{frame}

\begin{frame}{Stage 1 (2): Use overlapping pairs}
\bluekey{We need the required pairs to be observable, not all items at once.}
\begin{block}{Supported blocks}
Each required block has positive observation probability on its target support.
Use anchor pair $\{a,b\}$ and extension pairs $\{j,a\}$:
\[
\underbrace{(Y_a,Y_b,W)}_{\text{anchor law}}
\qquad\longleftrightarrow\qquad
\underbrace{(Y_j,Y_a,W)}_{\text{extension law for each }j}.
\]
\end{block}
\begin{itemize}
\item Apply Stage 1 separately to each pair, conditional on $X$.
\item The shared item $Y_a$ connects the recovered pair laws.
\item $Z_S$ corrects missingness; $W$ supplies variation for latent decomposition.
\end{itemize}
\medskip
\begin{takeaway}
\bluekey{Output: overlapping pair laws.}\quad
\redkey{Global complete cases are not required.}
\end{takeaway}
\end{frame}

\begin{frame}{Stage 2 (1): Identify the latent structure}
\bluekey{Decompose one anchor tensor, then recover every extension item.}\par
\smallskip
Let $F\in\{1,\ldots,r\}$ denote the latent class,
$M_j(y,f;x)=\Prb(Y_j=y\mid F=f,X=x)$, and
$\lambda_f(w,x)=\Prb(F=f\mid W=w,X=x)$.
\begin{block}{\textcolor{Red}{New!} Theorem 1: Key result}
The recovered law of $(Y_a,Y_b,W)\mid X$ identifies the anchor factors.
Overlapping extension laws then identify
\[
\underbrace{Q_F=\{\lambda_f(w,x)\}}_{\text{latent-class distribution}}
\qquad\text{and}\qquad
\underbrace{Q_Y=\{M_j(y,f;x)\}_j}_{\text{all measurement kernels}}.
\]
\end{block}
\begin{itemize}
\item Allman et al. (2009, \emph{AoS}): three views, with $W$ as the third mode.
\item Known finite $r$, finite supports, local independence and measurement exclusion.
\item Three-view ranks satisfy $k_a+k_b+k_G\ge 2r+2$ (Assumption 3.1).
\item Anchor ordering fixes \redkey{common labels}; otherwise identification is up to permutation.
\end{itemize}
\end{frame}

\begin{frame}{Stage 2 (2): Connect identification to estimation}
\begin{block}{Recover the joint outcome law}
For any item set $A$, including items never jointly observed,
\[
\Prb(Y_A=y_A\mid W=w,X=x)
=\sum_{f=1}^r\lambda_f(w,x)\prod_{j\in A}M_j(y_j,f;x).
\]
\end{block}
\begin{itemize}
\item \textbf{Stage 1:} estimate each pair's inverse observation probability.
\item \textbf{Stage 2:} fit a \term{bridge-weighted composite criterion} to the observed pairs.
\end{itemize}
\medskip
\begin{takeaway}
\bluekey{New: connect MNAR block-law recovery and latent decomposition in one identification theorem.}
\end{takeaway}
\medskip
\textbf{Next:} finite-dimensional inference under Appendix regularity conditions;
account for bridge estimation.
\end{frame}

\begin{frame}{\textcolor{Red}{New!} Theorem 5.1}
\begin{theorembox}{Consistency and asymptotic normality}
Under Assumption 5.1 and Proposition 4.1,
\[
\widehat\theta\to_p\theta_0,\qquad
\sqrt n(\widehat\theta-\theta_0)\rightsquigarrow N(0,A^{-1}BA^{-\top}).
\]
Let $\psi$ denote the composite score. At $(\theta_0,\eta_0)$,
\[
A=\E(\partial_\theta\psi),\qquad C=\E(\partial_\eta\psi),\qquad
B=\operatorname{Var}\{\psi+C\varphi_\eta\}.
\]
\end{theorembox}
\begin{itemize}
  \item The sandwich variance includes \textbf{first-stage bridge estimation}.
  \item Individual-level bootstrap repeats both stages and label alignment.
  \item A growing sieve needs additional rate and inverse-problem conditions.
\end{itemize}
\end{frame}

\begin{frame}{Simulation design: five estimators}
\begin{enumerate}
\item \textbf{Complete-data oracle:} product-mixture likelihood before missingness.
\item \textbf{MAR:} observed-data likelihood under ignorability.
\item \textbf{Correct selection likelihood:} true logistic selection model.
\item \textbf{Misspecified selection likelihood:} omit $Y_1$ main effect and interaction. Both selection fits exclude $Z_S$.
\item \textbf{Proposed finite-cell bridge:}
\[
\E\{R_Sr_S(Y_S)\mid Z_S,W\}=1.
\]
Direct normalized IPW, then weighted latent-class fitting.
\end{enumerate}
\bluekey{Proposed outcome recovery uses Stage 1; comparators use fitted models.}
\end{frame}

\begin{frame}{Contribution \& Limitation \& Future Work}
\bluekey{Contribution}
\begin{itemize}
\item Connect Missing-IV block-law recovery and latent decomposition in one theorem.
\item Identify class shares and measurement kernels under common labels; construct a bridge-weighted estimator.
\item No parametric selection model or global complete cases required.
\end{itemize}
\redkey{Limitation}
\begin{itemize}
\item Requires known finite $r$, supported pairs, exclusion, completeness, local independence, rank and anchor ordering.
\item Not robust to all model violations; not uniformly more efficient than correctly specified selection likelihood.
\end{itemize}
\bluekey{Future Work}
\begin{itemize}
\item Extend finite-dimensional inference to growing-sieve bridges with additional inverse-problem theory.
\end{itemize}
\end{frame}

\begin{frame}{Motivation and objective}
Let $D$ denote the response pattern and $W$ a latent shifter. Even if
\[
D\indep W\mid(Y,F,X),
\]
marginalizing $F$ gives
\[
\Prb(D=d\mid y,w,x)=\int\Prb(D=d\mid y,f,x)\,
\textcolor{Red}{d\Prb(f\mid y,w,x)}.
\]
The posterior may depend on $w$: $D\indep W\mid(Y,X)$ does \redkey{not} follow.
\medskip
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\textbf{$Z_S$: Missing IV}\\
$R_S\indep Z_S\mid(Y_S,U_S)$; predicts $Y_S$.\\
\bluekey{Stage 1: recover the block law.}
\end{column}
\begin{column}{.48\textwidth}
\textbf{$W$: latent shifter}\\
Shifts $F$; supplies third-mode variation.\\
\bluekey{Stage 2: separate $Q_F$ and $Q_Y$.}
\end{column}
\end{columns}
\medskip
\bluekey{Objective: connect both stages for joint identification of $Q_F,Q_Y$.}\\
No parametric selection model; finite classes, rank and label conditions.
\end{frame}

\begin{frame}{Assumption 1}
\begin{block}{Observation process}
Let $Y=(Y_1,\ldots,Y_m)$ denote the measurement vector and let
$D=(D_1,\ldots,D_m)\in\{0,1\}^m$ denote its response indicators.
\begin{itemize}
  \item $Y_j$ is observed if and only if $D_j=1$.
  \item $X,W,Z,D$ are always observed; the latent variable $F$ is not.
  \item $Z$ collects the block-specific Missing IVs $Z_S$.
\end{itemize}
The \term{observed data for one individual} are
\[
O=(X,W,Z,D,Y_D),\qquad Y_D=\{Y_j:D_j=1\}.
\]
\end{block}
\bluekey{$O$ records the response pattern and only the measurements actually observed.}
\end{frame}

\begin{frame}{Assumptions 2 and 3}
\begin{block}{Assumption 2: Latent-distribution shift}
Let $X$ denote ordinary covariates and $W$ the \term{latent shifter}.
The conditional latent distribution, with $Z$ marginalized, is
\[
F\mid(W,X)\sim Q_F(\cdot\mid W,X).
\]
\end{block}
\begin{block}{Assumption 3: Measurement process}
Let $Q_j$ denote the \term{measurement kernel} of item $j$ and
$\theta_j$ its parameter. For $j=1,\ldots,m$,
\[
Y_j\mid(F,X)\sim Q_j(\cdot\mid F,X;\theta_j).
\]
\end{block}
\bluekey{$Q_F$ describes who is in each latent class; $Q_j$ describes what each class measures.}
\end{frame}

\begin{frame}{Assumption 4}
\begin{block}{Local independence and measurement exclusion}
With $Z$ marginalized, the \textbf{joint} measurement law satisfies
\[
P(Y\mid F,W,X)=\prod_{j=1}^m Q_j(Y_j\mid F,X).
\]
\begin{itemize}
  \item All items are mutually conditionally independent given $(F,X)$.
  \item $W$ shifts $Q_F$ but does not directly enter the measurement kernels.
  \item This is a restriction on the full item vector, not merely pairwise independence.
\end{itemize}
\end{block}
\begin{takeaway}
We do not assume $Y\indep Z\mid(F,W,X)$. Missing-IV exclusion is a separate restriction on the response indicator (Assumption 6).
\end{takeaway}
\end{frame}

\begin{frame}{Lemma 2.1}
\small
Let $U=(X,W)$, $R=\prod_{j=1}^mD_j$, and $P_1=\Prb(Y,U\mid R=1)$.
$U,Z,R$ are always observed; the entire vector $Y$ is observed when $R=1$.
\begin{theorembox}{Global complete-case recovery}
Assume $R\indep Z\mid(Y,U)$, $\pi(y,u)=\Prb(R=1\mid y,u)>0$, and $\pi^{-1}\in L^2(P_1)$.
For every $h\in L^2(P_1)$, assume
\[
\E\{h(Y,U)\mid R=1,Z,U\}=0\ \Longrightarrow\ h=0\quad P_1\text{-a.s.}
\]
Then $\pi$ uniquely solves, among positive $q$ with $q^{-1}\in L^2(P_1)$,
\[
\E\{R/q(Y,U)\mid Z,U\}=1,
\]
For every integrable $h$, $\E h(Y,Z,U)=\E\{Rh(Y,Z,U)/\pi(Y,U)\}$ identifies the full law.
\end{theorembox}
\bluekey{A direct vector extension, but it requires global complete-case positivity.}
\par{\footnotesize d'Haultfoeuille (2010, \emph{JoE}); Miao et al. (2019, arXiv).}
\end{frame}

\begin{frame}{Assumption 5}
\begin{block}{Supported block}
A subset $S\subseteq\{1,\ldots,m\}$ is \term{supported} when all measurements in $Y_S$ are jointly observed with positive probability on the target support. A global complete case is not required.
\end{block}

$U_S,Z_S,R_S$ are always observed. Use $Y_S$ as a complete block only when $R_S=1$.

\begin{frame}{Assumptions 6 and 7}
\begin{block}{Assumption 6: Blockwise covariate-adjusted Missing IV}
For every supported block $S$,
\[
R_S\indep Z_S\mid(Y_S,U_S).
\]
\end{block}
\begin{block}{Assumption 7: Complete-case completeness}
For $h\in L^2\{\Prb(Y_S,U_S\mid R_S=1)\}$,
\[
\E\{h(Y_S,U_S)\mid R_S=1,Z_S,U_S\}=0
\Longrightarrow h(Y_S,U_S)=0.
\]
The conclusion holds almost surely on $\{R_S=1\}$.
\end{block}
Positivity transfers null sets between the complete-case and full laws; completeness makes the observed conditional operator injective.
\end{frame}

\begin{frame}{\textcolor{Red}{New!} Proposition 1}
\small
\begin{theorembox}{Supported-block identification}
Under Assumptions 1, 5--7, let $P_{1S}=\Prb(Y_S,U_S\mid R_S=1)$ and assume $\pi_S^{-1}\in L^2(P_{1S})$.
Then $\pi_S$ is the unique positive solution among $q$ with $q^{-1}\in L^2(P_{1S})$ of
\[
\E\left\{\frac{R_S}{q(Y_S,U_S)}\middle|Z_S,U_S\right\}=1.
\]
Consequently, for every integrable $h$,
\[
\E h(Y_S,Z_S,U_S)
=\E\left\{\frac{R_Sh(Y_S,Z_S,U_S)}{\pi_S(Y_S,U_S)}\right\},
\]
and $\Prb(Y_S,Z_S,U_S)$ is identified.
\end{theorembox}
\bluekey{Uniqueness holds in the stated bridge class; the true propensity supplies existence.}
\end{frame}

\begin{frame}{The remaining decomposition problem}
\textbf{After Stage 1:} each required block law is known; $F$ is still unobserved.
Fix $X=x$ and $F\in\{1,\ldots,r\}$.
\[
\begin{aligned}
p_f(x)&=\Prb(F=f\mid X=x),&
G(w,f;x)&=\Prb(W=w\mid F=f,X=x),\\
M_j(y,f;x)&=\Prb(Y_j=y\mid F=f,X=x).
\end{aligned}
\]
Let $\theta=(p,G,M_1,\ldots,M_m)$ and $\Phi_x:\theta\mapsto T(\theta)$.
\[
P_{\mathrm{obs},\theta}=P_{\mathrm{obs},\widetilde\theta}
\quad\Longrightarrow\quad T(\theta)=T(\widetilde\theta)
\qquad\text{(Proposition 1).}
\]
\begin{itemize}
\item Injectivity up to permutation identifies $[\theta]\in\Theta_x/\mathfrak S_r$.
\item Anchor ordering fixes labels and identifies $\theta$ itself.
\end{itemize}
\bluekey{Targets: $Q_F=\{\lambda_f(w,x)\}$ and $Q_Y=\{M_j\}$.}
{\slideannotation Identification terminology: Lewbel (2019, \emph{JEL}).}
\end{frame}

\begin{frame}{Assumption 3.1: Support and model}
\begin{block}{Covariate-assisted anchor-pair structure (1/2)}
Let $F\in\{1,\ldots,r\}$ with known $r\ge2$, and let the item spaces
$\mathcal Y_j$ and the support $\mathcal W_x$ of $W\mid X=x$ be finite.
For every $x$ in a common set $\mathcal X_0$ with $P_X(\mathcal X_0)=1$:
\begin{itemize}
  \item Assumptions 2--4 hold; $p_f(x)>0$ for every class, and $P(W=w\mid X=x)>0$ for every $w\in\mathcal W_x$.
  \item The anchor pair $\{a,b\}$ and every extension pair $\{j,a\}$, $j\notin\{a,b\}$, satisfy Assumptions 1 and 5--7 for every $w$.
  \item Each required pair satisfies the bridge-class condition of Proposition 1:
  \[
  \pi_S^{-1}\in L^2\{P(Y_S,U_S\mid R_S=1)\}.
  \]
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Assumption 3.1: Rank and labels}
\bluekey{The same three-view Kruskal condition as Allman et al. (2009, \emph{AoS}, Theorem 1).}
\begin{block}{Covariate-assisted anchor-pair structure (2/2)}
Let $k_a,k_b,k_G$ be the column Kruskal ranks of $M_a,M_b,G$.
For every $x\in\mathcal X_0$,
\[
\textcolor{Red}{\bm{k_a+k_b+k_G\ge 2r+2}}.
\]
For a known anchor score $s_a$, define
\[
\mu_{af}(x)=\sum_y s_a(y)M_a(y,f;x).
\]
The same strict ordering $\mu_{a1}(x)<\cdots<\mu_{ar}(x)$ holds throughout $\mathcal X_0$.
\end{block}
\bluekey{The rank condition identifies the factors; anchor ordering fixes a common class label.}
\end{frame}

\begin{frame}{Lemma 3.1}
\begin{theorembox}{Khatri--Rao rank}
Let $A$ and $B$ have $r$ nonzero columns, with column Kruskal ranks $k_A$ and $k_B$. Then
\[
k_{B\odot A}\ge \min\{k_A+k_B-1,r\}.
\]
In particular,
\[
\redkey{$k_A+k_B\ge r+1$}
\quad\Longrightarrow\quad
\bluekey{$B\odot A$ has full column rank.}
\]
\end{theorembox}
\begin{takeaway}
This lemma converts the anchor-tensor rank condition into the linear-inversion rank needed for every extension item.
\end{takeaway}
\end{frame}

\begin{frame}{\textcolor{Red}{New!} Theorem 1}
\begin{theorembox}{Anchor-pair identification with $W$ as the third mode}
Under the stage-1 assumptions and the anchor-pair conditions, the observed-data law identifies, under the fixed anchor ordering,
\[
p(x),\quad G(x),\quad
Q_F(x)=\{\lambda_f(w,x)\}_{f,w},\quad
Q_Y(x)=\{M_j(y,f;x)\}_{j,y,f}.
\]
Moreover, for any set of items $A$,
\[
\Prb(Y_A=y_A\mid W=w,X=x)
=\sum_{f=1}^r\lambda_f(w,x)\prod_{j\in A}M_j(y_j,f;x)
\]
is identified even if all items in $A$ are never jointly observed.
\end{theorembox}
\vspace{.2em}
\bluekey{In short: recover the pair laws, decompose one anchor tensor, and obtain the latent-class distribution and every measurement kernel.}
\end{frame}


\begin{frame}{Bridge-weighted composite estimation}
\small
Stage 1 follows d'Haultfoeuille (2010, \emph{JoE}, Section 3): solve the conditional moment equation blockwise and recover each pair law by IPW. Stage 2 follows the plug-in order of Zhao and Shao (2015, \emph{JASA}, Sections 2.2--2.3), while fitting one latent mixture to all overlapping pair laws.
\begin{columns}[T,onlytextwidth]
\begin{column}{.57\textwidth}
Let $\mathcal S_*=\{\{a,b\}\}\cup\{\{j,a\}:j\notin\{a,b\}\}$ and
\[
p_\theta(y_S,w\mid x)=\sum_{f=1}^r p_f(x)G(w,f;x)
\prod_{j\in S}M_j(y_j,f;x).
\]
The proposed estimator is
\[
\widehat\theta=\arg\max_{\theta\in\Theta}
\frac1n\sum_{i=1}^n\sum_{S\in\mathcal S_*}c_S
\frac{R_{iS}\log p_\theta(Y_{iS},W_i\mid X_i)}
{\widehat\pi_S(Y_{iS},U_i)}.
\]
\end{column}
\begin{column}{.40\textwidth}
\begin{itemize}
  \item \term{Composite M-estimator}, not a full likelihood.
  \item The theorem covers a finite-dimensional bridge.
  \item A growing sieve requires additional inverse-problem theory.
  \item $c_S>0$ balances contributions across blocks.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Conditions for estimation}
\begin{block}{Proposition 4.1: A unique population maximizer}
\begin{itemize}
  \item Assumptions 1--7 and 3.1 hold, including anchor ordering.
  \item $\theta_0\in\Theta$ correctly generates the latent measurement model.
  \item For each required pair, $p_\theta(y_S,w\mid x)>0$ on a common support.
  \item The population criterion uses the true bridge and fixed weights $c_S>0$.
\end{itemize}
Then the bridge-weighted population criterion has the unique maximizer $\theta_0$.
\end{block}
\bluekey{Bridge weighting recovers full pair expectations; zero KL loss fixes every pair law, and Theorem 1 fixes the latent parameters.}
\end{frame}

\begin{frame}{Assumption 5.1: Regularity}
Let $O_1,\ldots,O_n$ be i.i.d. and write $\beta=(\theta,\eta)$,
where $\eta=(\eta_S:S\in\mathcal S_*)$ contains the bridge parameters.
Use free coordinates after removing simplex constraints.
\begin{block}{Regularity and separation (1/3)}
\begin{itemize}
  \item $\theta,\eta$ are finite-dimensional and correctly specified.
  \item Their parameter spaces are compact; the true parameters are interior points.
  \item The first-stage population moment has the unique zero $\eta_0$.
  \item The criterion and first-stage moments are twice continuously differentiable near the truth, with square-integrable envelopes.
  \item The bridge Jacobian, composite-score Jacobian, and corresponding Godambe matrix are nonsingular.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Assumption 5.1: Uniform convergence}
\begin{block}{Regularity and separation (2/3)}
Let $\widehat Q_n(\theta,\widehat\eta)$ denote the plug-in criterion and
$Q_0(\theta)$ its population version with the true bridge.
\[
\sup_{\theta\in\Theta}
|\widehat Q_n(\theta,\widehat\eta)-Q_0(\theta)|\xrightarrow{p}0.
\]
The population maximum is separated: for every $\varepsilon>0$ with a nonempty constraint set,
\[
\sup_{\theta\in\Theta:\|\theta-\theta_0\|\ge\varepsilon}Q_0(\theta)
<Q_0(\theta_0).
\]
\end{block}
\bluekey{These global conditions justify consistency; local smoothness alone does not.}
\end{frame}

\begin{frame}{Assumption 5.1: First stage and separation}
\begin{block}{Regularity and separation (3/3)}
The first-stage estimator is regular and asymptotically linear:
\[
\sqrt n(\widehat\eta-\eta_0)
=n^{-1/2}\sum_{i=1}^n\varphi_\eta(O_i)+o_p(1),\quad
\E\varphi_\eta=0,\quad \E\|\varphi_\eta\|^2<\infty.
\]
The following are uniformly bounded away from zero:
\begin{itemize}
  \item class probabilities and supported-pair propensities;
  \item the smallest nonzero singular value of the anchor tensor's $W$-mode unfolding (rows: $W$; columns: $(Y_a,Y_b)$);
  \item the minors supporting the Kruskal-rank conditions;
  \item adjacent gaps in the anchor ordering.
\end{itemize}
\end{block}
\bluekey{Identification alone does not imply regular estimation or stable inference.}
\end{frame}

\begin{frame}{Appendix: Proof of Theorem 1 (Steps 1--2)}
\begin{block}{Step 1: supported-pair laws}
New Proposition 1 identifies
\[
P(Y_a,Y_b,W,X),\qquad P(Y_j,Y_a,W,X),
\]
and therefore $\mathcal T_{abW}(x)$ and every $\mathcal T_{jaW}(x)$.
\end{block}
\begin{block}{Step 2: three-view latent-class representation}
Local independence gives
\[
\mathcal T_{abW}(x)
=\sum_{f=1}^r p_f(x)
M_a(:,f;x)\otimes M_b(:,f;x)\otimes G(:,f;x).
\]
\end{block}
\begin{takeaway}
$Y_a$, $Y_b$, and $W$ are the three conditionally independent views.
\end{takeaway}
\end{frame}

\begin{frame}{Appendix: Proof of Theorem 1 (Steps 3--4)}
\begin{block}{Step 3: Kruskal uniqueness}
With column Kruskal ranks $k_a,k_b,k_G$,
\[
\textcolor{Red}{\bm{k_a+k_b+k_G\ge 2r+2}}
\]
identifies $M_a,M_b,\widetilde G$ up to a common permutation and column scaling, where $\widetilde G(:,f)=p_fG(:,f)$.
\end{block}
\begin{block}{Step 4: scaling and labels}
Probability-column normalization gives
\[
p_f=\bm 1^\top\widetilde G(:,f),\qquad G(:,f)=\widetilde G(:,f)/p_f.
\]
The anchor ordering $\mu_{a1}<\cdots<\mu_{ar}$ fixes the common class labels.
\end{block}
\end{frame}

\begin{frame}{Appendix: Proof of Theorem 1 (Steps 5--6)}
\begin{block}{Step 5: rank for linear inversion}
The anchor Kruskal condition and Lemma 3.1 imply
\[
G\odot M_a\quad\text{has full column rank}.
\]
Hence the extension-tensor equation has a unique right-inverse solution.
\end{block}
\begin{block}{Step 6: latent class distribution given the shifter}
Because $p$, $G$, and $\Prb(W=w\mid X=x)$ are identified, Bayes' rule gives
\[
\lambda_f(w,x)
=\frac{p_f(x)G(w,f;x)}{\Prb(W=w\mid X=x)}.
\]
Thus $Q_F(x)=\{\lambda_f(w,x)\}_{f,w}$ is identified.
\end{block}
\end{frame}

\begin{frame}{Appendix: Proof of Theorem 1 (Steps 7--8)}
\begin{block}{Step 7: extension-kernel inversion}
For every nonanchor item $j$,
\[
M_j=T_{j\mid aW}
\left[\operatorname{diag}(p)(G\odot M_a)^\top\right]^+,
\]
so all measurement kernels $Q_Y=\{M_j\}_{j=1}^m$ are identified.
\end{block}
\begin{block}{Step 8: joint laws never observed together}
For any item set $A$,
\[
\Prb(Y_A=y_A\mid W=w,X=x)
=\sum_{f=1}^r\lambda_f(w,x)\prod_{j\in A}M_j(y_j,f;x).
\]
This includes sets $A$ that are never jointly observed.
\end{block}
\end{frame}

\begin{frame}{Appendix: empirical diagnostics and falsification checks}
For a proposed assignment of blockwise IVs, latent shifters, covariates, and anchor items, the data can diagnose:
\begin{itemize}
  \item relevance of each $Z_S$ and rank of a finite complete-case operator;
  \item supported-block positivity and effective sample size;
  \item third-mode rank, anchor separation, and label stability;
  \item agreement of measurement kernels recovered from redundant blocks.
\end{itemize}
\begin{takeaway}
Failure can refute a design, but success does not verify shadow exclusion, measurement exclusion, or local independence.
\end{takeaway}
\end{frame}
